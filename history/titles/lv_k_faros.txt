c_vehar = {
    600.1.1 = { holder = Local_Rulers_Vahar }
    7899.1.1 = {
        government = feudal_government # Doing feudal first instead of pirate govt prevents the unhandled succession errors that pop up
		effect = {
			if = {
				limit = {
					exists = holder
					holder = { NOT = { has_realm_law = confederate_partition_succession_law } }
				}
				holder = {
					add_realm_law = confederate_partition_succession_law
				}
			}
		}
    }
}
