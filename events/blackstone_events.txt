blackstone.3 = {
    type = character_event
    title = blackstone.3_t
    desc = {
        first_valid = {

        }
    } 

    theme = learning

    right_portrait = {
        character = root
        animation = survey
    }

    option = {
        name = active.01
        involved_activity = {
            if = {
                limit = {
                    has_activity_option = { # Do this if you selected the scholar option
                        category = special
                        option = study_type_scholarly
                    }
                }
                province:3499 = { # Province 3499 is the Citadel
                    save_scope_as = study_destination
                }
            }
            else_if = {
                limit = {
                    has_activity_option = { # Do this if you selected the religion option
                        category = special
                        option = study_type_faith
                    }
                    faith = { any_holy_site = {} } # Gotta make sure the faith has any holy sites first
                }
                faith = {
                    random_holy_site = { # Pick any holy random site of our faith to go to
                        title_province = { save_scope_as = study_destination }
                    }
                }
            }
            else_if = {
                limit = {
                    has_activity_option = { # Do this if you selected the rock licker option
                        category = special
                        option = study_type_excavation
                    }
                }
                random_county = {
                    limit = { # Must be any of the blackstone counties
                        OR = {
                            this = province:9763
                            this = province:9470
                            this = province:9434
                            this = province:9618
                            this = province:9150
                            this = province:8223
                            this = province:9520
                            this = province:7351
                            this = province:4415
                        }
                    }
                    title_province = { save_scope_as = study_destination }
                }
            }
            else_if = { # For the occult one we can polish this more later, but for now let's make it the same as the faith
                limit = {
                    has_activity_option = { # Do this if you selected the occult option
                        category = special
                        option = study_type_occult
                    }
                    faith = { any_holy_site = {} } # Gotta make sure the faith has any holy sites first
                }
                faith = {
                    random_holy_site = { # Pick any holy random site of our faith to go to
                        title_province = { save_scope_as = study_destination }
                    }
                }
            }
        }
        start_travel_plan = {
			destination = scope:study_destination
			return_trip = yes
			on_arrival_event = blackstone.4
		}
    }
}
