# Overridden from agot_colonization_events.txt
namespace = agot_colonization_events

## Called from yearly events to check if AI should colonize
## For human player version check scripted_guis/agot_wilderness.txt -> gui_colonize
agot_colonization_events.0006 = {
	hidden = yes
	scope = character

	trigger = {
		### Basic checks
		is_ai = yes
		is_landed = yes

		NOT = { any_owned_story = { story_type = story_conqueror } } #Conquerors have their own logic

		trigger_if = {
			limit = { is_independent_ruler = no }
			has_character_flag = vassal_directive_settle_wilderness
		}

		### Culture/Character ID triggers
		agot_is_dummy_character = no

		### Blocking Governments
		NOT = { government_has_flag = government_is_nw }
		NOT = { government_has_flag = government_is_uninteractable }

		### Check whether the ruler is above their settlement limit
		above_settlement_limit = no

		### Not on cooldown
		NOT = { has_variable = colonization_cooldown }

		### Only surrounding lands
		any_sub_realm_county = {
			any_title_to_title_neighboring_and_across_water_county = {
				holder = { government_has_flag = government_is_wilderness }
				NOT = { has_county_modifier = block_settlement_ability }
				#BTW Blocks
				trigger_if = {
					limit = {
						title_province ?= {
							geographical_region = world_westeros_beyond_the_wall
						}
					}
					root = {
						OR = {
							culture = { agot_is_wildling_culture = yes }
							capital_county.title_province ?= {
								geographical_region = world_westeros_beyond_the_wall
							}
						}
					}
				}
				# LEGACY OF VALYRIA ADDED
				save_temporary_scope_as = ruin
			}
		}

		### Character Income Trigger
		# LEGACY OF VALYRIA MODIFIED
		trigger_if = {
			limit = { has_game_rule = lv_colonisation_agot }
			trigger_if = { # If count
				limit = { highest_held_title_tier = tier_county }
				gold >= 50
				monthly_character_income >= 2
			}
			trigger_else_if = { # If duke
				limit = { highest_held_title_tier = tier_duchy }
				gold >= 75
				monthly_character_income >= 1.75
			}
			trigger_else_if = { # If king
				limit = { highest_held_title_tier = tier_kingdom }
				gold >= 100
				monthly_character_income >= 1.75
			}
			trigger_else_if = { # If emporer
				limit = { highest_held_title_tier = tier_empire }
				gold >= 125
				monthly_character_income >= 1.5
			}
			trigger_else = { gold >= 150 } # Fallback
		}
		trigger_else = {
			# Legacy of Valyria has its own cost calculation
			# Requires a ruin scope, hence why this was moved here at the bottom
			# gold >= lv_settlement_cost
			# REVISE SOON
			gold >= 1000
			NOR = {
				var:lv_settlement_state ?= flag:planning
				any_owned_story = {
					story_type = lv_settlement_story_cycle
					is_target_in_variable_list = {
						name = lv_settlement_expeditions
						target = scope:ruin
					}
				}
			}
		}
	}
	immediate = {
		debug_log = "agot_colonization_events.0006: AI considering colonization..."

		every_sub_realm_county = {
			every_title_to_title_neighboring_and_across_water_county = {
				limit = {
					holder = { government_has_flag = government_is_wilderness }
					NOR = {
						has_county_modifier = block_settlement_ability
						holder = root
					}
					#BTW Blocks
					trigger_if = {
						limit = {
							title_province ?= {
								geographical_region = world_westeros_beyond_the_wall
							}
						}
						root = {
							OR = {
								culture = { agot_is_wildling_culture = yes }
								capital_county.title_province ?= {
									geographical_region = world_westeros_beyond_the_wall
								}
							}
						}
					}

					# LEGACY OF VALYRIA ADDED
					trigger_if = {
						limit = { title_province ?= { geographical_region = world_valyria_mainland } }
						# Subregion must already be green, implying someone insane enough (mainly just the player) proved it is possible
						lv_valyria_subregion_restored = yes
					}
					trigger_else = { always = yes }
				}
				add_to_list = adjacent_wilderness
			}
		}

		ordered_in_list = {
			list = adjacent_wilderness
			order_by = {
				value = 0
				every_neighboring_county = {
					limit = { holder = root }
					add = 2
				}
				every_neighboring_county = {
					limit = {
						holder = root
						duchy = prev.duchy
					}
					add = 16
				}
				every_neighboring_county = {
					limit = {
						holder = root
						kingdom = prev.kingdom
					}
					add = 8
				}
				every_neighboring_county = {
					limit = {
						holder = root
						empire = prev.empire
					}
					add = 4
				}
			}
			save_scope_as = wilderness
		}

		### MAKE AI COLONISE COUNTY ###
		# LEGACY OF VALYRIA MODIFIED
		if = {
			limit = { has_game_rule = lv_colonisation_agot }
			scope:wilderness ?= { debug_log_scopes = yes } # Only needed for testing
			if = {
				limit = {
					exists = scope:wilderness
				}
				ai_colonization_effect = { WILDERNESS = scope:wilderness ROOT_SCOPE = root }
			}
		}
		else = {
			scope:wilderness ?= { save_scope_as = ruin }
			save_scope_as = settler
			set_variable = {
				name = lv_settlement_expedition_liege
				value = scope:liege
			}
			set_variable = {
				name = lv_settlement_expedition_destination
				value = scope:ruin
			}
			scope:liege = {
				set_variable = {
					name = lv_settlement_cost
					value = scope:lv_settlement_cost
				}
				remove_variable = lv_settlement_state
				if = {
					limit = { owns_story_of_type = lv_settlement_story_cycle }
					random_owned_story = {
						limit = { story_type = lv_settlement_story_cycle }
						add_to_variable_list = {
							name = lv_settlement_expeditions
							target = scope:ruin
						}
					}
				}
				else = { create_story = lv_settlement_story_cycle }
			}
			start_travel_plan = {
				destination = scope:ruin
				players_use_planner = yes
				return_trip = no
				can_cancel_planning = yes
				on_start_on_action = lv_settlement_on_start
				on_arrival_on_action = lv_settlement_on_arrival
			}
		}
	}
}
