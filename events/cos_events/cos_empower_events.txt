namespace = cos_empower

# Empower using piety
cos_empower.1001 = {
	type = character_event
	title = cos_empower.1001.t
	desc = cos_empower.1001.desc

	theme = stewardship_domain_focus
	override_background = { reference = sothoryos_the_maw }

	left_portrait = {
		character = root
		animation = chancellor
	}

	option = {
		name = cos_empower.1001.a
		add_piety = -250
		every_county = {
				limit = {
					faith = faith:the_eternal_maw	
				}
				add_county_modifier = { 
					modifier = cos_empowered_maw_1
					years = 1
				}
		}
	}

	option = {
		name = cos_empower.1001.b
		add_piety = -500
		every_county = {
				limit = {
					faith = faith:the_eternal_maw	
				}
				add_county_modifier = { 
					modifier = cos_empowered_maw_2
					years = 1
				}
		}
	}

	option = {
		name = cos_empower.1001.c
		add_piety = -750
		every_county = {
				limit = {
					faith = faith:the_eternal_maw	
				}
				add_county_modifier = { 
					modifier = cos_empowered_maw_3
					years = 1
				}
		}
		
	}
}
