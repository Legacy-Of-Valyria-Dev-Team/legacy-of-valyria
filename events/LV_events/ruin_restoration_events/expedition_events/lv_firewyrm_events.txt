# Firewyrm encounter
lv_firewyrm_event.0001 = {
	type = character_event
	title = lv_firewyrm_event.0001.t
	desc = lv_firewyrm_event.0001.desc

	window = agot_dragon_character_event
	theme = dragon

	override_background = {
		trigger = {
			scope:firewyrm_source = {
				lv_valyria_subregion_restored = yes
				OR = {
					agot_is_desert_terrain = yes
					agot_is_desert_mountains_terrain = yes
				}
			}
		}
		reference = ep2_travel_settlement_ruin_desert
	}
	override_background = {
		trigger = {
			scope:firewyrm_source = {
				lv_valyria_subregion_restored = yes
				OR = {
					agot_is_drylands_terrain = yes
					agot_is_oasis_terrain = yes
					agot_is_canyon_terrain = yes
				}
			}
		}
		reference = drylands
	}
	override_background = {
		trigger = {
			scope:firewyrm_source = {
				lv_valyria_subregion_restored = yes
				OR = {
					agot_is_steppe_terrain = yes
				}
			}
		}
		reference = mpo_steppe_evening
	}
	override_background = {
		trigger = {
			scope:firewyrm_source = {
				lv_valyria_subregion_restored = yes
				OR = {
					agot_is_taiga_terrain = yes
					agot_is_taiga_bog_terrain = yes
					agot_is_glacier_terrain = yes
					agot_is_frozen_flats_terrain = yes
				}
			}
		}
		reference = bp3_steppe_winter
	}
	override_background = {
		trigger = { scope:firewyrm_source = { lv_valyria_subregion_restored = yes } }
		reference = plains
	}
	override_background = {
		trigger = { scope:firewyrm_source = { lv_valyria_subregion_partially_restored = yes } }
		reference = lv_valyria_surface_partial_rehabilitation
	}
	override_background = {
		trigger = { scope:firewyrm_source = { lv_valyria_subregion_ruined = yes } }
		reference = lv_valyria_surface
	}
	override_effect_2d = {
		trigger = { scope:firewyrm_source = { lv_valyria_subregion_ruined = yes } }
		reference = smoke
	}

	left_portrait = {
		character = root
		camera = camera_human_back_to_camera
		scripted_animation = duel_wield_weapon
	}

	right_portrait = {
        character = scope:firewyrm
		animation = dragon_main
		camera = lv_camera_firewyrm_event_right

		triggered_outfit = {
			outfit_tags = {
				dynamic_camera_zoom_0.8
				no_clothes
				bald
			}
		}
    }

	widgets = {
		widget = {
			gui = "agot_event_shader_sparks"
			container = "agot_shader_event_foreground"
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:firewyrm } }
			if = {
				limit = { NOT = { exists = scope:firewyrm_source } }
				location = { save_scope_as = firewyrm_source }
			}
			lv_spawn_firewyrm_effect = { LOCATION = scope:firewyrm_source }
		}
		if = {
			limit = {
				has_trait = dragonrider
				any_relation = { type = agot_dragon is_alive = yes }
			}
			random_relation = {
				type = agot_dragon
				limit = {
					has_trait = dragon
					has_variable = current_rider
					any_relation = { type = agot_dragon }
				}
				save_scope_as = dragon
			}
		}
	}

	option = { # Fuck it, we ball (with dragon)
		name = lv_firewyrm_event.0001.duel.dragon
		trigger = {
			has_trait = dragonrider
			exists = scope:dragon
		}
		custom_tooltip = lv_firewyrm_event.0001.duel.dragon.tt
		lv_start_firewyrm_duel_combat_effect = {
			ATTACKER = root
			DEFENDER = scope:firewyrm
		}
	}

	option = { # Fuck it, we ball
		name = lv_firewyrm_event.0001.duel
		custom_tooltip = lv_firewyrm_event.0001.duel.tt
		lv_slay_firewyrm_effect = {
			SLAYER = root
			FIREWYRM = scope:firewyrm
		}
	}

	option = {
		name = lv_firewyrm_event.0001.flee
		custom_tooltip = lv_firewyrm_event.0001.flee.tt
		#scope:firewyrm = { death = { death_reason = death_monster_duel } }
	}
}

		#}
		widget = {
			gui = "agot_event_shader_sparks"
			container = "agot_shader_event_foreground"
		}
	}

	immediate = {
		scope:firewyrm_location = {
			if = {
				limit = { NOT = { has_holding_type = wilderness_holding } }
				county = { agot_dragon_damage_county_effect = yes }
				if = {
					limit = { has_holding_type = settlement_holding }
					random_list = {
						80 = { lv_downgrade_settlement_progress_effect = yes }
						20 = { # Decimate
							county = {
								if = {
									limit = { development_level >= 1 }
									change_development_level = -1
								}
								else = { make_settlement_county_wilderness = { COUNTY = this } }
							}
						}
					}
				}
				else_if = {
					limit = { has_holding_type = ruin_holding }
					if = {
						limit = { has_building_or_higher = small_ruin_01 }
						if = {
							limit = { has_building = small_ruin_04 }
							remove_building = small_ruin_04
							hidden_effect = { add_building = small_ruin_03 }
						}
						else_if = {
							limit = { has_building = small_ruin_03 }
							remove_building = small_ruin_03
							hidden_effect = { add_building = small_ruin_02 }
						}
						else_if = {
							limit = { has_building = small_ruin_02 }
							remove_building = small_ruin_02
							hidden_effect = { add_building = small_ruin_01 }
						}
					}
					else_if = {
						limit = { has_building_or_higher = medium_ruin_01 }
						if = {
							limit = { has_building = medium_ruin_04 }
							remove_building = medium_ruin_04
							hidden_effect = { add_building = medium_ruin_03 }
						}
						else_if = {
							limit = { has_building = medium_ruin_03 }
							remove_building = medium_ruin_03
							hidden_effect = { add_building = medium_ruin_02 }
						}
						else_if = {
							limit = { has_building = medium_ruin_02 }
							remove_building = medium_ruin_02
							hidden_effect = { add_building = medium_ruin_01 }
						}
					}
					else_if = {
						limit = { has_building_or_higher = large_ruin_01 }
						if = {
							limit = { has_building = large_ruin_04 }
							remove_building = large_ruin_04
							hidden_effect = { add_building = large_ruin_03 }
						}
						else_if = {
							limit = { has_building = large_ruin_03 }
							remove_building = large_ruin_03
							hidden_effect = { add_building = large_ruin_02 }
						}
						else_if = {
							limit = { has_building = large_ruin_02 }
							remove_building = large_ruin_02
							hidden_effect = { add_building = large_ruin_01 }
						}
					}
				}
				else = {
					county = {
						if = {
							limit = { development_level >= 1 }
							change_development_level = -1
						}
						else = { agot_ruin_county_effect = yes }
					}
				}
			}
		}
	}

	option = { name = lv_firewyrm_event.1000.a }
}

# Process round
lv_firewyrm_duel.0002 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				OR = {
					scope:fd_attacker.var:fd_chance_of_winning >= 100
					scope:fd_defender = { is_alive = no }
				}
			}
			save_scope_value_as = {
				name = fd_finished
				value = yes
			}
			scope:fd_attacker = { save_scope_as = fd_victor }
			scope:fd_defender = { save_scope_as = fd_loser }
		}
		if = {
			limit = {
				OR = {
					scope:fd_defender.var:fd_chance_of_winning >= 100
					scope:fd_attacker = { is_alive = no }
					scope:fd_attacker_dragon = { is_alive = no }
				}
			}
			save_scope_value_as = {
				name = fd_finished
				value = yes
			}
			scope:fd_defender = { save_scope_as = fd_victor }
			scope:fd_attacker = { save_scope_as = fd_loser }
		}
		if = {
			limit = { exists = scope:fd_finished }
			finalise_firewyrm_duel_combat_results_effect = yes
		}
		# If we don't, and this round is at or over the round_cap_limit, enter into sudden death.
		else_if = {
			limit = { var:fd_combat_current_round >= 10 }
			if = {
				limit = { scope:fd_attacker_dragon.dragon_combat_effectiveness > scope:fd_defender.dragon_combat_effectiveness }
				save_scope_value_as = {
					name = fd_finished
					value = yes
				}
				scope:fd_attacker = { save_scope_as = fd_victor }
				scope:fd_defender = { save_scope_as = fd_loser }
			}
			else_if = {
				limit = { scope:fd_defender.dragon_combat_effectiveness > scope:fd_attacker_dragon.dragon_combat_effectiveness }
				save_scope_value_as = {
					name = fd_finished
					value = yes
				}
				scope:fd_defender = { save_scope_as = fd_victor }
				scope:fd_attacker = { save_scope_as = fd_loser }
			}
			# Otherwise, their prowess must be even, so we do a toss-up.
			else = {
				random_list = {
					50 = {
						save_scope_value_as = {
							name = fd_finished
							value = yes
						}
						scope:fd_attacker = { save_scope_as = fd_victor }
						scope:fd_defender = { save_scope_as = fd_loser }
					}
					50 = {
						save_scope_value_as = {
							name = fd_finished
							value = yes
						}
						scope:fd_defender = { save_scope_as = fd_victor }
						scope:fd_attacker = { save_scope_as = fd_loser }
					}
				}
			}
			finalise_firewyrm_duel_combat_results_effect = yes
		}
		else = {
			change_variable = {
				name = fd_combat_current_round
				add = 1
			}
			scope:fd_attacker = { trigger_event = lv_firewyrm_duel.0001 }
		}
	}
}
