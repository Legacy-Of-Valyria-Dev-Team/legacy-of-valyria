# Process round
lv_firewyrm_duel.0002 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				OR = {
					scope:fd_dragonrider.var:fd_chance_of_winning >= 100
					scope:fd_firewyrm = { is_alive = no }
				}
			}
			save_scope_value_as = {
				name = fd_finished
				value = yes
			}
			scope:fd_dragonrider = { save_scope_as = fd_victor }
			scope:fd_firewyrm = { save_scope_as = fd_loser }
		}
		if = {
			limit = {
				OR = {
					scope:fd_firewyrm.var:fd_chance_of_winning >= 100
					scope:fd_dragonrider = { is_alive = no }
					scope:fd_dragon = { is_alive = no }
				}
			}
			save_scope_value_as = {
				name = fd_finished
				value = yes
			}
			scope:fd_firewyrm = { save_scope_as = fd_victor }
			scope:fd_dragonrider = { save_scope_as = fd_loser }
		}
		if = {
			limit = { exists = scope:fd_finished }
			finalise_firewyrm_vs_dragon_combat_results_effect = yes
		}
		# If we don't, and this round is at or over the round_cap_limit, enter into sudden death.
		else_if = {
			limit = { var:fd_combat_current_round >= round_cap_limit }
			if = {
				limit = { scope:fd_dragon.dragon_combat_effectiveness > scope:fd_firewyrm.dragon_combat_effectiveness }
				save_scope_value_as = {
					name = fd_finished
					value = yes
				}
				scope:fd_dragonrider = { save_scope_as = fd_victor }
				scope:fd_firewyrm = { save_scope_as = fd_loser }
			}
			else_if = {
				limit = { scope:fd_firewyrm.dragon_combat_effectiveness > scope:fd_dragon.dragon_combat_effectiveness }
				save_scope_value_as = {
					name = fd_finished
					value = yes
				}
				scope:fd_firewyrm = { save_scope_as = fd_victor }
				scope:fd_dragonrider = { save_scope_as = fd_loser }
			}
			# Otherwise, their prowess must be even, so we do a toss-up.
			else = {
				random_list = {
					50 = {
						save_scope_value_as = {
							name = fd_finished
							value = yes
						}
						scope:fd_dragonrider = { save_scope_as = fd_victor }
						scope:fd_firewyrm = { save_scope_as = fd_loser }
					}
					50 = {
						save_scope_value_as = {
							name = fd_finished
							value = yes
						}
						scope:fd_firewyrm = { save_scope_as = fd_victor }
						scope:fd_dragonrider = { save_scope_as = fd_loser }
					}
				}
			}
			finalise_firewyrm_vs_dragon_combat_results_effect = yes
		}
		else = {
			change_variable = {
				name = fd_combat_current_round
				add = 1
			}
			trigger_event = lv_firewyrm_duel.0001
		}
	}
}
