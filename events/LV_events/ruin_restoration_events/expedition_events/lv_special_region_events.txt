namespace = lv_special_region_event

# First time setting foot in major ruined region
lv_special_region_event.0001 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_proper } }
				desc = lv_special_region_event.0001.t.valyria
			}
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_lands_of_the_long_summer } }
				desc = lv_special_region_event.0001.t.lands_of_the_long_summer
			}
			triggered_desc = {
				trigger = { location = { geographical_region = sea_smoking_sea } }
				desc = lv_special_region_event.0001.t.smoking_sea
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_proper } }
				desc = {
					desc = lv_special_region_event.0001.desc.valyria
					first_valid = {
						triggered_desc = { # At least 50% of Valyria must be reclaimed
							trigger = {
								any_county_in_region = { # TO-DO
									region = world_valyria_proper
									percent >= 0.5
									title_province = { NOT = { has_holding_type = ruin_holding } }
								}
							}
							desc = lv_special_region_event.0001.desc.valyria.restored
						}
						desc = lv_special_region_event.0001.desc.valyria.ruined
					}
				}
			}
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_lands_of_the_long_summer } }
				desc = {
					desc = lv_special_region_event.0001.desc.lands_of_the_long_summer
					first_valid = {
						triggered_desc = { # At least 50% of the Lands of the Long Summer must be reclaimed
							trigger = {
								any_county_in_region = { # TO-DO
									region = world_valyria_lands_of_the_long_summer
									percent >= 0.5
									title_province = { NOT = { has_holding_type = ruin_holding } }
								}
							}
							desc = lv_special_region_event.0001.desc.lands_of_the_long_summer.restored
						}
						desc = lv_special_region_event.0001.desc.lands_of_the_long_summer.ruined
					}
				}
			}
			triggered_desc = {
				trigger = { location = { geographical_region = sea_smoking_sea } }
				desc = {
					desc = lv_special_region_event.0001.desc.smoking_sea
					first_valid = {
						triggered_desc = { # A certain percentages of coastal volcanoes must no longer be ruins
							trigger = { # Alternatively, a certain percentage of coastal counties must no longer have noxious gases
								any_county_in_region = { # TO-DO
									region = world_valyria_lands_of_the_long_summer
									percent >= 0.5
									title_province = { NOT = { has_holding_type = ruin_holding } }
								}
							}
							desc = lv_special_region_event.0001.desc.smoking_sea.restored
						}
						desc = lv_special_region_event.0001.desc.smoking_sea.ruined
					}
				}
			}
		}
	}
	theme = travel

	# Valyria
	override_background = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
				}
				OR = {
					agot_is_urban_terrain = yes
					has_building_or_higher = large_ruined_holding_01
				}
			}
		}
		reference = lv_valyria_large
	}
	override_background = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
				}
				has_building_or_higher = medium_ruined_holding_01
			}
		}
		reference = lv_valyria_medium
	}
	override_background = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
				}
			}
		}
		reference = lv_valyria_small
	}

	override_effect_2d = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
					geographical_region = sea_smoking_sea
				}
			}
		}
		reference = smoke
	}

	left_portrait = {
		character = root
		animation = stunned
	}

	option = {
		name = {
			trigger = { location = { geographical_region = world_valyria_proper } }
			text = lv_special_region_event.0001.a.valyria
		}
		name = {
			trigger = { location = { geographical_region = world_valyria_lands_of_the_long_summer } }
			text = lv_special_region_event.0001.a.lands_of_the_long_summer
		}
		name = {
			trigger = { location = { geographical_region = sea_smoking_sea } }
			text = lv_special_region_event.0001.a.smoking_sea
		}
	}
}

# Smoking Sea
lv_special_region_event.0002 = {
	type = character_event
	trigger = {
		is_available_travelling_adult = yes
		location = {
			OR = {
				travel_danger_type = {
					travel_plan = root.current_travel_plan
					terrain = smoking_sea
				}
				travel_danger_type = {
					travel_plan = root.current_travel_plan
					terrain = smoking_sea_volcano
				}
			}
		}
	}
	title = lv_special_region_event.0001.t.smoking_sea
	desc = {
		desc = lv_special_region_event.0001.desc.smoking_sea
		first_valid = {
			triggered_desc = { # A certain percentages of coastal volcanoes must no longer be ruins
				trigger = { # Alternatively, a certain percentage of coastal counties must no longer have noxious gases
					any_county_in_region = { # TO-DO
						region = world_valyria_lands_of_the_long_summer
						percent >= 0.5
						title_province = { NOT = { has_holding_type = ruin_holding } }
					}
				}
				desc = lv_special_region_event.0001.desc.smoking_sea.restored
			}
			desc = lv_special_region_event.0001.desc.smoking_sea.ruined
		}
	}
	theme = travel

	# Valyria
	override_background = { reference = lv_valyria_smoking_sea }

	left_portrait = {
		character = root
		animation = stunned
	}

	widgets = {
		widget = {
			gui = "agot_event_shader_sparks"
			container = "agot_shader_event_foreground"
		}
	}

	option = { name = lv_special_region_event.0001.a.smoking_sea }
}
