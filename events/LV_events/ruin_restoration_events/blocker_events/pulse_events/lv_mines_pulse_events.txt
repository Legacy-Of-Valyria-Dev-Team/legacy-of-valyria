namespace = lv_mines_pulse

# Directory
# lv_mines_pulse.0001 - Annual firewyrm pulse event
# lv_mines_pulse.0002 - Firewyrm event proceeds, choose who to send

# Annual firewyrm pulse event
# - This event will provide a random chance of either flavour text or actually spawning a firewyrm
lv_mines_pulse.0001 = {
	type = character_event
	title = lv_mines_pulse.0001.t
	desc = lv_mines_pulse.0001.desc
	theme = dragon

	override_background = { reference = lv_valyria_surface }
	override_effect_2d = { reference = smoke }

	immediate = {
		scope:mines_province = {
			random_list = {
				# Uncomment and finalize chances only once all firewyrm-related stuff are done
				10 = {
					set_variable = {
						name = lv_mines_outcome
						value = flag:firewyrm
					}
				}
				90 = {}
			}
		}
	}

	option = { # Firewyrm pops out and makes lair, you're toast
		name = lv_mines_pulse.0001.a
		if = {
			limit = { scope:mines_province = { var:lv_mines_outcome ?= flag:firewyrm } }
			scope:mines_province = { save_scope_as = firewyrm_source }
			custom_tooltip = {
				text = lv_mines_pulse.0001.a.tt.firewyrm
				trigger_event = {
					id = lv_firewyrm_event.0001
					days = { 7 28 }
				}
			}
		}
		else = { custom_tooltip = lv_mines_pulse.0001.a.tt }
	}

	after = {
		scope:mines_province = {
			if = {
				limit = { has_variable = lv_mines_outcome }
				remove_variable = lv_mines_outcome
			}
		}
	}
}
