namespace = lv_valyrian_freehold

# Directory
# lv_valyrian_freehold.0001 - Form Valyrian Freehold
# lv_valyrian_freehold.0002 - Notification: Announce formation of the Freehold

# Form Valyrian Freehold
lv_valyrian_freehold.0001 = {
	type = character_event
	window = fullscreen_event
	title = lv_valyrian_freehold.0001.t
	desc = lv_valyrian_freehold.0001.desc
	theme = emperor

	override_background = { reference = lv_valyria_prospering }

	immediate = { save_scope_as = restorer }

	option = {
		name = lv_valyrian_freehold.0001.a
		set_nickname_effect = { NICKNAME = nick_the_restorer }
		if = {
			limit = { government_has_flag = government_is_feudal }
			custom_tooltip = {
				text = lv_valyrian_freehold.0001.tooltip.styling
				title:h_valyrian_freehold = {
					set_variable = {
						name = lv_freehold_style
						value = flag:imperial
					}
				}
			}
		}
		lv_restore_valyrian_freehold_effect = yes
	}

	after = {
		every_player = {
			limit = { this != scope:restorer }
			trigger_event = lv_valyrian_freehold.0002
		}
	}
}

# Notification: Announce formation of the Freehold
lv_valyrian_freehold.0002 = {
	type = character_event
	window = fullscreen_event
	title = lv_valyrian_freehold.0002.t
	desc = lv_valyrian_freehold.0002.desc
	theme = emperor

	override_background = { reference = lv_valyria_prospering }

	immediate = {
		save_scope_as = restorer
		custom_tooltip = lv_valyrian_freehold.0002.tooltip
	}

	option = { name = lv_valyrian_freehold.0002.a }
}
