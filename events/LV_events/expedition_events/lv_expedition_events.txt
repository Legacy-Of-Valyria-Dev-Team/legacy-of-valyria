# lv_expedition_event.0010 - Arrival

# Decide whether to lead the expedition yourself or assign an expedition leader
lv_expedition_event.0001 = {
	type = character_event
	title = lv_expedition_event.0001.t
	desc = lv_expedition_event.0001.desc
	theme = travel

	right_portrait = {
		character = root
		animation = thinking
	}

	left_portrait = {
		character = scope:new_leader
		animation = thinking
	}

	immediate = {
		random_courtier = {
			save_scope_as = new_leader
		}
	}

	# Personally lead the expedition - Proceed to next phase
	option = {
		name = lv_expedition_event.0001.a
		current_travel_plan = { resume_travel_plan = yes }
	}

	# Choose a courtier to be the expedition leader - Set up story cycle for selected courtier then end activity, assign story cycle instead
	option = {
		name = lv_expedition_event.0001.b
		#skip_activity_phase = yes
		current_travel_plan = {
			travel_plan_activity = { set_activity_host = scope:new_leader }
		}
	}
}

namespace = lv_special_region_event

# First time setting foot in major ruined region
lv_special_region_event.0001 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_proper } }
				desc = lv_special_region_event.0001.t.valyria
			}
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_lands_of_the_long_summer } }
				desc = lv_special_region_event.0001.t.lands_of_the_long_summer
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_proper } }
				desc = {
					desc = lv_special_region_event.0001.desc.valyria
					first_valid = {
						triggered_desc = { # At least 50% of Valyria must be reclaimed
							trigger = {
								any_county_in_region = {
									region = world_valyria_proper
									percent >= 0.5
									title_province = { NOT = { has_holding_type = ruin_holding } }
								}
							}
							desc = lv_special_region_event.0001.desc.valyria.restored
						}
						desc = lv_special_region_event.0001.desc.valyria.ruined
					}
				}
			}
			triggered_desc = {
				trigger = { location = { geographical_region = world_valyria_lands_of_the_long_summer } }
				desc = {
					desc = lv_special_region_event.0001.desc.lands_of_the_long_summer
					first_valid = {
						triggered_desc = { # At least 50% of the Lands of the Long Summer must be reclaimed
							trigger = {
								any_county_in_region = {
									region = world_valyria_lands_of_the_long_summer
									percent >= 0.5
									title_province = { NOT = { has_holding_type = ruin_holding } }
								}
							}
							desc = lv_special_region_event.0001.desc.lands_of_the_long_summer.restored
						}
						desc = lv_special_region_event.0001.desc.lands_of_the_long_summer.ruined
					}
				}
			}
		}
	}
	theme = travel

	# Valyria
	override_background = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
				}
				OR = {
					agot_is_urban_terrain = yes
					has_building_or_higher = large_ruined_holding_01
				}
			}
		}
		reference = lv_valyria_large
	}
	override_background = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
				}
				has_building_or_higher = medium_ruined_holding_01
			}
		}
		reference = lv_valyria_medium
	}
	override_background = {
		trigger = {
			location = {
				OR = {
					geographical_region = world_valyria_proper
					geographical_region = world_valyria_lands_of_the_long_summer
				}
			}
		}
		reference = lv_valyria_small
	}

	override_effect_2d = {
