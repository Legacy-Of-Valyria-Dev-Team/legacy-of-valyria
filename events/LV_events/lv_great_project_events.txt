# Notification: First attempt
lv_great_project.0001 = {
	type = character_event
	title = lv_great_project.0001.t
	desc = {
		first_valid = {
			triggered_desc = { # Rehabilitation has worked before and it's your first time doing it
				trigger = {
					global_var:lv_world_valyria_rehabilitation_status ?= flag:succeeded
					NOT = { exists = var:lv_world_valyria_attempted_rehabilitation }
				}
				desc = lv_great_project.0001.desc.succeeded
			}
			triggered_desc = { # Rehabilitation has been attempted but has yet to succeed
				trigger = { global_var:lv_world_valyria_rehabilitation_status ?= flag:attempted }
				desc = lv_great_project.0001.desc.attempted
			}
			desc = lv_great_project.0001.desc
		}
	}
	theme = dragon

	left_portrait = {
		character = root
		animation = idle
	}

	override_background = {
		trigger = { scope:province = { lv_valyria_subregion_restored = yes } }
		reference = plains
	}
	override_background = {
		trigger = { scope:province = { lv_valyria_subregion_partially_restored = yes } }
		reference = lv_valyria_surface_partial_rehabilitation
	}
	override_background = {
		trigger = { scope:province = { lv_valyria_subregion_ruined = yes } }
		reference = lv_valyria_surface
	}

	override_effect_2d = {
		trigger = { scope:province = { lv_valyria_subregion_ruined = yes } }
		reference = smoke
	}

	option = {
		name = {
			trigger = {
				global_var:lv_world_valyria_rehabilitation_status ?= flag:succeeded
				NOT = { exists = var:lv_world_valyria_attempted_rehabilitation }
			}
			text = lv_great_project.0001.a.succeeded
		}
		name = {
			trigger = { global_var:lv_world_valyria_rehabilitation_status ?= flag:attempted }
			text = lv_great_project.0001.a.attempted
		}
		name = {
			trigger = { NOT = { exists = global_var:lv_world_valyria_rehabilitation_status } }
			text = lv_great_project.0001.a
		}
	}

	after = {
		if = {
			limit = { NOT = { exists = global_var:lv_world_valyria_rehabilitation_status } }
			set_global_variable = {
				name = lv_world_valyria_rehabilitation_status
				value = flag:attempted
			}
		}
		set_variable = {
			name = lv_world_valyria_attempted_rehabilitation
			value = yes
		}
	}
}
