# Background event to make sure we're in Valyria before the next ghost event, since it takes 200 days
lv_valyria_random_event.0011 = {
    hidden = yes

    immediate = {
        if = { # If any subregion of Valyria still ruined/valid, proceed with the check
            limit = { situation:valyria_situation = { any_situation_sub_region = { NOT = { has_sub_region_phase_parameter = valyria_green } } } }
            if = {
                limit = {
                    location = {
                        geographical_region = world_valyria_mainland
                        lv_valyria_subregion_ruined = yes
                    }
                }
                trigger_event = lv_valyria_random_event.0020
            }
            else = {
                trigger_event = { # Let's evaluate every 7 days
                    id = lv_valyria_random_event.0011
                    days = 7
                }
            }
        }
        # Else do nothing, unless you want some sort of special/secret event about "silencing the ghosts" or something
    }
}

            20 = { add_gold = 75 }         # captured a few alive for experimentation

        }
    }
    option = {

    }
