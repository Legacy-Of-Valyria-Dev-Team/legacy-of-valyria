lv_valyria_random_event.01 = {
    title = lv_valyria_random_event.01.t
    desc = lv_valyria_random_event.01.desc
    theme = dread
    override_background = { reference = lv_valyria_surface }
    left_portrait = {
		character = root
		animation = paranoia
	}
    trigger = {
        location = {
			OR = {
				geographical_region = world_valyria_proper
				geographical_region = world_valyria_lands_of_the_long_summer
			}
		}
        NOT = { has_character_flag = saw_figures }
    }
	
	immediate = { add_character_flag = saw_figures }


	option = {
        name = lv_valyria_random_event.01.a
        add_stress = 25
        add_trait = paranoid
		trigger_event = lv_valyria_random_event.02 days = 200  # ~7 months later
    }
    option = {
        name = lv_valyria_random_event.01.b
        add_prestige = 100
        add_stress = 40
		trigger_event = lv_valyria_random_event.02 days = 200  # ~7 months later
    }
    option = {
        name = lv_valyria_random_event.01.c
        add_stress = -20
        add_trait = lunatic_1
		add_character_flag = lv_is_lunatic
        trigger_event = lv_valyria_random_event.02 days = 200  # ~7 months later
    }
}
