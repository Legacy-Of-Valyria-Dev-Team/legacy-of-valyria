# Upgrading Redwyrm's Wrath
lv_special_blockers.0003 = {
    type = character_event
    title = lv_special_blockers.0003.t
    desc = lv_special_blockers.0003.desc
    theme = realm

    right_portrait = {
        character = root
        animation = personality_rational
    }

    artifact = {
        target = scope:newly_created_artifact
        position = lower_left_portrait
    }

    override_background = { reference = lov_vault_01 }

    immediate = {
        hidden_effect = {
            remove_list_global_variable = {
                name = lv_unavailable_unique_items
                target = flag:torment_sword_lost
            }
            lv_create_artifact_vs_torment_effect = {
                OWNER = root
            }
        }
    }

    option = {
        name = lv_special_blockers.0003.a
        custom_tooltip = gain_vs_artifact
        add_gold = 250
        add_prestige = 1000
        dynasty = {
            add_dynasty_prestige = 500
        }
    }
}
