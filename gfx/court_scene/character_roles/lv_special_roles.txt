sitting_heir = {
	effect = {
		scope:ruler = {
			if = {
				limit = {
					exists = primary_heir
					primary_heir = {
						is_available_healthy_adult = yes
					}
				}
				primary_heir = { add_to_list = characters }
			}
		}
	}

	# spouse sits on the throne, so this animation parameter is ignored
	# instead we use animation specific to the throne entity
	# see game/gfx/court_scene/throne_animations/throne_animations.txt
	scripted_animation = {
		animation = throne_room_ruler
	}
}

triarchy_magister = {
	effect = {
		scope:ruler = {
			every_vassal = {
				limit = {
					is_available_healthy_adult = yes
					OR = {
						this = title:k_tyrosh.holder
						this = title:k_myr.holder
						this = title:k_lys.holder
					}
				}
				add_to_list = characters
			}
		}
	}

	# spouse sits on the throne, so this animation parameter is ignored
	# instead we use animation specific to the throne entity
	# see game/gfx/court_scene/throne_animations/throne_animations.txt
	scripted_animation = {
		animation = throne_room_ruler
	}
}

triarch = {
	effect = {
		scope:ruler = {
			every_vassal = {
				limit = {
					is_available_healthy_adult = yes
					OR = {
						has_character_flag = third_triarch
						has_character_flag = second_triarch
						has_character_flag = dragon_council
					}
				}
				add_to_list = characters
			}

					is_available_healthy_adult = yes
