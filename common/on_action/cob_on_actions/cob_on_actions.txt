on_game_start = {
	on_actions = {
		cob_on_game_start
		cob_set_legacies
		cob_house_dna
		cob_disputed_lands_cleanup
		cob_set_dragonstone_faiths
	}
}

on_game_start_after_lobby = {
	on_actions = {
		cob_on_game_start_after_lobby
	}
}

cob_on_game_start = {
	effect = {
		cv_add_camp_buildings_on_start = yes
	}
}

cob_on_game_start_after_lobby = {
	effect = {
		if = {
			limit = {
				NOR = {
					has_game_rule = agot_shatter_duchies
					has_game_rule = agot_shatter_duchies_even
					has_game_rule = agot_shatter_petty_kingdoms
					has_game_rule = agot_shatter_kingdoms
					has_game_rule = agot_shatter_empires
				}
			}
			if = { #LoV added - manage 7899.7.22 Century of Blood Scenerio
				limit = { game_start_date = 7899.7.22 }

				lv_7899_7_22_scenario_setup = yes
			}
		}
	}
}
