cob_on_game_start = {
    effect = {
        cv_add_camp_buildings_on_start = yes
        cob_character_appearance_setup = yes
        if = {
            limit = {
                game_start_date < 7902.1.1
            }
            agot_set_legitimate_house = {
                TITLE = title:e_the_riverlands
                HOUSE = character:Teague_1.house
            }
        }
        title:e_the_stormlands.holder ?= {
            if = {
                limit = { has_title = title:e_the_riverlands }
                set_primary_title_to = title:e_the_stormlands
            }
        }
		character:Varezys_1 = {
			set_father = character:Varezys_cob_39
			set_mother = character:Varezys_cob_40
		}
    }
}

restore_valyria_cob_event_on_actions = {
	trigger = {
		has_character_flag = varezys_valyria_event
	}
	effect = {
		if = {
			limit = {
				character:Varezys_1 ?= {
					has_character_flag = varezys_valyria_event
					OR = {
						location = province:10325
						location = province:10344
						location = province:10484
						location = province:10388
						location = province:10383
						location = province:10378
						location = province:10448
						location = province:10437
						location = province:10206
						location = province:10209
					}
				}
			}
			if = {
				limit = {
					AND = {
						is_ai = yes
						location = province:10206
					}
				}
				if = {
					limit = { exists = character:dragon_vassarion }
					character:dragon_vassarion = {
						death = { death_reason = death_vanished }
					}
				}
				death = { death_reason = death_vanished }
			}
			else = {
				delay_travel_plan = {
					days = 30
					add = yes
				}
				trigger_event = {
					on_action = restore_valyria_cob_travel_on_actions
				}
			}
		}
	}
}

aurion_armor_event_on_actions = {
    effect = {
        if = {
            limit = {
                root ?= {
                    location = { has_building = aurions_lost_expedition }
                    is_target_in_global_variable_list = {
                        name = lv_unavailable_unique_items
                        target = flag:aurion_sword_lost
                    }
                    is_target_in_global_variable_list = {
                        name = lv_unavailable_unique_items
                        target = flag:aurion_armor_lost
                    }
                }
            }
            trigger_event = {
                id = cob_7899_scenerio.9998
            }
        }
    }
}
