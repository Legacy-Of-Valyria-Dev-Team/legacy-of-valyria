cob_on_game_start = {
	effect = {
		cv_add_camp_buildings_on_start = yes
		if = {
			limit = {
				game_start_date < 7902.1.1
			}
			agot_set_legitimate_house = {
				TITLE = title:k_the_riverlands
				HOUSE = character:Teague_1.house
			}
		}
	}
}

cob_on_game_start_after_lobby = {
	effect = {
		if = {
			limit = {
				NOR = {
					has_game_rule = agot_shatter_duchies
					has_game_rule = agot_shatter_duchies_even
					has_game_rule = agot_shatter_petty_kingdoms
					has_game_rule = agot_shatter_kingdoms
					has_game_rule = agot_shatter_empires
				}
			}
			if = { #LoV added - manage 7899.6.17 Century of Blood Scenerio
				limit = { game_start_date = 7899.6.17 }

				cob_7899_7_22_scenario_setup = yes
			}
		}
	}
}
