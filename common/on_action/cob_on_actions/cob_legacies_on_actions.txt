			dynasty:dynn_Sonaryen = { agot_set_dynasty_level_history_effect = { LEVEL = 6 } }

			dynasty:dynn_Belaerys = { agot_set_dynasty_level_history_effect = { LEVEL = 6 } }

cob_brightroar_global_flag = {
	effect = {
		if = {
			limit = {
				game_start_date < 7900.1.1
			}
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:recovered_brightroar
			}
		}
		if = {
			limit = {
				game_start_date > 7900.1.1
			}
			add_to_global_variable_list = {
				name = lv_unavailable_unique_items
				target = flag:aurion_sword_lost
			}
			add_to_global_variable_list = {
				name = lv_unavailable_unique_items
				target = flag:aurion_armor_lost
			}
		}
	}
}
