			dynasty:dynn_Mataeryon = { agot_set_dynasty_level_history_effect = { LEVEL = 5 } }

			dynasty:dynn_Benaris = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }

			dynasty:dynn_Paenohrin = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
