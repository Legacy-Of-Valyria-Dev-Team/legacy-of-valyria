# AGOT Essos Expansion Government On Actions
# All realm/government-related on_actions goes here
on_game_start = {
    on_actions = {
		agot_EE_on_start
    }
}

agot_EE_on_start = {
    effect = {
		# Dynamically set special placeholder or EE-added cultural traditions to already-existing cultures
		# This prevents us from using direct file overrides if we don't have to and is generally cleaner for cases such as AGOT itself updating those cultures
		culture:mantaryan = {
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_mantaryan } }
				add_culture_tradition = lv_tradition_mantaryan
			}
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_north_valyrian_1 } }
				add_culture_tradition = lv_tradition_north_valyrian_1
			}
		}
		culture:tolosi = {
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_tolosi } }
				add_culture_tradition = lv_tradition_tolosi
			}
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_north_valyrian_1 } }
				add_culture_tradition = lv_tradition_north_valyrian_1
			}
		}
		culture:elyrian = {
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_elyrian } }
				add_culture_tradition = lv_tradition_elyrian
			}
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_north_valyrian_1 } }
				add_culture_tradition = lv_tradition_north_valyrian_1
			}
		}

		# Ghiscari cultures
		culture:astapori = { if = { limit = { NOT = { has_cultural_tradition = agot_tradition_astapori } } add_culture_tradition = agot_tradition_astapori } }

		# Some Blue stuff
		# Dothraki culture
		culture:dothraki = { if = { limit = { NOT = { has_cultural_tradition = lv_tradition_the_stallions } } add_culture_tradition = lv_tradition_the_stallions } }

		# Others
		culture:others_culture = { if = { limit = { NOT = { has_cultural_tradition = lv_tradition_the_others } } add_culture_tradition = lv_tradition_the_others } }
		culture:others_culture = { if = { limit = { NOT = { has_cultural_tradition = tradition_winter_warriors } } add_culture_tradition = tradition_winter_warriors } }
		culture:others_culture = { if = { limit = { NOT = { has_cultural_tradition = tradition_collective_lands } } add_culture_tradition = tradition_collective_lands } }
		culture:others_culture = { if = { limit = { NOT = { has_cultural_tradition = tradition_mountain_homes } } add_culture_tradition = tradition_mountain_homes } }
		culture:others_culture = { if = { limit = { NOT = { has_cultural_tradition = tradition_quarrelsome } } add_culture_tradition = tradition_quarrelsome } }

		# Give Old Blood modifiers to Volantene dynasties
        dynasty:dynn_Maegyr = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
		dynasty:dynn_Vhassar = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Paenymion = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Qhaedar = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Vaelaros = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Staegone = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Tagaros = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Nogarys = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Votar = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Qoheros = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
	}
}
