#Use this to add extra buildings to the tribal lords making them stronger. They should be a bitch to fight on their own terms.
cos_tribal_buffs_game_start_pulse = {
	effect = {
		every_province = { # Maw Tribe Holdings
			limit = { has_building = the_maw_2 }
			add_building = palisades_02
			add_building = war_camps_02
			add_building = longhouses_02
			add_building = market_villages_02
		}
		every_province = { # Blood Cultists
			limit = { has_building = bloodletters_hall_2 }
			add_building = palisades_02
			add_building = war_camps_02
			add_building = longhouses_02
			add_building = market_villages_02
		}
		every_province = { # Glade Tribe
			limit = { has_building = balam_glade_2 }
			add_province_modifier = extra_building_slot
			add_province_modifier = extra_building_slot
			add_building = longhouses_02
			add_building = market_villages_02
		}
		every_province = { # Mound Tribe
			limit = { has_building = cos_corpse_mound_2 }
			add_building = palisades_02
			add_building = war_camps_02
		}
	}
}

cos_on_game_start_after_lobby = {
	effect = {
	
		##### Scenario setup start #####
		
		if = {
			limit = {
				NOR = {
					has_game_rule = agot_shatter_duchies
					has_game_rule = agot_shatter_duchies_even
					has_game_rule = agot_shatter_petty_kingdoms
					has_game_rule = agot_shatter_kingdoms
					has_game_rule = agot_shatter_empires
				}
			}
			if = {
				limit = {
					title:c_the_maw = {
					exists = this
					}
				}
				title:c_the_maw.holder = {
					trigger_event = cos_start_quests.1000
				}
			}
			cos_start_scenario = yes
		}
	}
}
