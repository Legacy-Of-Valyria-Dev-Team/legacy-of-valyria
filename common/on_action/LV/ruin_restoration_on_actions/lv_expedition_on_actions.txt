# Legacy of Valyria Expedition on actions

# Root: Character
travel_danger_events_on_action = {
	on_actions = {
		lv_travel_danger_events_on_action
		lv_travel_danger_firewyrm_on_action
	}
}

lv_travel_danger_events_on_action = {
	trigger = { is_migrating = no }
	random_events = {
		chance_of_no_event = {
			value = {
				add = 100
				if = {
					limit = {
						exists = root.current_travel_plan
						exists = root.location
					}
					subtract = root.current_travel_plan.current_danger_value
					add = root.current_travel_plan.travel_safety
				}
			}
			max = {
				value = 100
				subtract = define:NTravel|TRAVEL_DANGER_MINIMUM
			}
		}
		10 = 0
		350 = lv_travel_danger.0001 # Smoking Sea: The Fog Lives
		350 = lv_travel_danger.0002 # Smoking Sea: Blood on the Deck
	}
}

lv_travel_danger_firewyrm_on_action = {
	trigger = {
		is_migrating = no
		location = {
			NAND = {
				has_variable = lv_firewyrm
				has_building = firewyrm_lair
			}
			has_building_or_higher = valyria_mines_01
		}
	}
	effect = {
		random = {
			chance = 10
			# Consider adding a pre-event, e.g. an event warning you of something roaring or screeching
			trigger_event = lv_firewyrm_event.0001
		}
	}
}

on_travel_plan_movement = { on_actions = { lv_special_location_on_action } }
