# Legacy of Valyria Expedition on actions

# Forced Valyrian travel danger events
on_travel_plan_movement = {
	on_actions = {
		lv_travel_danger_events_on_action
		lv_travel_danger_sea_events_on_action
		lv_travel_danger_firewyrm_on_action
	}
}

# Valyrian
lv_travel_danger_events_on_action = {
	trigger = {
		is_migrating = no # No Danger events while migrating
		#AGOT Added
		NOR = {
			current_travel_plan ?= { has_travel_option = dragon_flight_option }
			has_trait = dragon
		}
		# If Valyria, must NOT be green
		location = { lv_valyria_subregion_ruined = yes }
	}
	random_events = {
		chance_of_no_event = {
			value = {
				add = 100
				if = {
					limit = {
						exists = root.current_travel_plan
						exists = root.location
					}
					subtract = root.current_travel_plan.current_danger_value
					add = root.current_travel_plan.travel_safety
				}
			}
			max = {
				value = 100
				subtract = define:NTravel|TRAVEL_DANGER_MINIMUM
			}
		}
		100 = 0

		50 = lv_valyria_random_event.0009 # Ghosts
		175 = lv_valyria_random_event.0040 # Dragon Sighting
		350 = lv_valyria_random_event.0050 # Ruins: The Vanishing City
		175 = lv_valyria_random_event.0090 # Earthquake reveals lava river
		175 = lv_valyria_random_event.0100 # Swarm
		175 = lv_valyria_random_event.0110 # Ash Serpents
		350 = lv_valyria_random_event.0120 # Toxic Fumes: Sulfur
	}
}

lv_travel_danger_sea_events_on_action = {
	trigger = {
		is_migrating = no # No Danger events while migrating
		#AGOT Added
		NOR = {
			current_travel_plan ?= { has_travel_option = dragon_flight_option }
			has_trait = dragon
		}
		# If Valyria, must NOT be green
		location = {
			trigger_if = {
				limit = { geographical_region = sea_smoking_sea }
				always = no # Turn off Smoking Sea danger events for now until we can make a GP for it
			}
			trigger_else = { always = yes }
		}
	}
	random_events = {
		chance_of_no_event = {
			value = {
				add = 100
				if = {
					limit = {
						exists = root.current_travel_plan
						exists = root.location
					}
					subtract = root.current_travel_plan.current_danger_value
					add = root.current_travel_plan.travel_safety
				}
			}
			max = {
				value = 100
				subtract = define:NTravel|TRAVEL_DANGER_MINIMUM
			}
		}
		10 = 0
		350 = lv_travel_danger.0001 # Smoking Sea: The Fog Lives
		350 = lv_travel_danger.0002 # Smoking Sea: Blood on the Deck
	}
}

lv_travel_danger_firewyrm_on_action = {
	trigger = {
		is_migrating = no # No Danger events while migrating
		#AGOT Added
		NOR = {
			current_travel_plan ?= { has_travel_option = dragon_flight_option }
			has_trait = dragon
		}
		location = {
			NAND = {
				has_variable = lv_firewyrm
				has_building = firewyrm_lair
			}
			has_building_or_higher = valyria_mines_01
			NOT = { has_building = valyria_mines_03 }
			# Maybe also include the Fourteen Flames, but only if Valyria is not green
			lv_valyria_subregion_ruined = yes
		}
	}
	effect = {
		random = {
			chance = 10
			# Consider adding a pre-event, e.g. an event warning you of something roaring or screeching
			location = { save_scope_as = firewyrm_source }
			trigger_event = lv_firewyrm_event.0001
		}
	}
}

					limit = { geographical_region = world_valyria_proper }

									NOT = { has_variable = lv_visited_valyria }

							}
