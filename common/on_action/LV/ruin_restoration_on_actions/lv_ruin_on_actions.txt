# Legacy of Valyria Ruin on actions
# - On actions to make rulers quit harassing the poor ruins

on_war_started = { on_actions = { lv_ruin_on_war_started } }

# STOP WARRING ON RUINS IDIOT AI
lv_ruin_on_war_started = {
	trigger = {
		OR = {
			scope:attacker = { has_trait = ruin }
			scope:defender = { has_trait = ruin }
		}
	}
	effect = { scope:war = { end_war = white_peace } }
}
