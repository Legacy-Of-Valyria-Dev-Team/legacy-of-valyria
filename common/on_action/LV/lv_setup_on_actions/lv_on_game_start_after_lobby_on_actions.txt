# Legacy of Valyria On Game Start After Lobby On Actions
# - All on actions effects after pressing play goes here
on_game_start_after_lobby = {
	on_actions = {
		lv_ruins_setup
		lv_ruins_setup_coa
		lv_post_start_fixes
	}
}

lv_ruins_setup = {
	trigger = {
		NOT = { has_game_rule = lv_colonisation_agot }
		NOT = { has_game_rule = lv_colonisation_instant }
	}
	effect = {
		# Set up blockers
		every_county_in_region = {
			limit = {
				title_province = {
					OR = {
						has_holding_type = ruin_holding
						has_holding_type = wilderness_holding
					}
				}
			}
			region = world_valyria_mainland
			title_province = {
				# PREDEFINED BLOCKERS
				if = {
					limit = { this = province:10256 }
					lv_add_blocker_effect = {
						FORCE = yes
						BLOCKER_BUILDING = lokirian_barracks_01
					}
				}
				if = {
					limit = { has_holding_type = ruin_holding }
					lv_blocker_generation_predefined_ruined_holdings_effect = yes
				}
				lv_blocker_generation_predefined_mines_effect = yes
				lv_blocker_generation_predefined_noxious_gases_effect = yes

				# RANDOMLY-GENERATED BLOCKERS
				if = {
					limit = { NOT = { has_game_rule = lv_colonisation_lite } }
					lv_blocker_generation_random_noxious_gases_effect = yes
					lv_blocker_generation_random_unexplored_caves_effect = yes
					lv_blocker_generation_random_swamp_effect = yes
					lv_blocker_generation_random_stonemen_encampment_effect = yes
					lv_blocker_generation_random_bandit_camps_effect = yes
				}
			}
		}
	}
}

lv_post_start_fixes = {
	effect = {
		# Remove dummy characters from confederations
		every_ruler ?= {
			confederation ?= {
				save_scope_as = lakefolk_confederation
				every_confederation_member = {
					limit = { agot_is_dummy_character = yes }
					scope:lakefolk_confederation = { remove_confederation_member = prev }
				}
			}
		}
	}
}
