# Legacy of Valyria On Game Start On Actions
# - All on actions effects after initial map generation (but before playing) goes here
on_game_start = {
    on_actions = {
		lv_remove_realms

		lv_culture_and_history_setup
		lv_provinces_setup
		lv_titles_setup
		lv_vassals_setup
		lv_on_game_start

		cob_on_game_start
		cob_set_legacies
		cob_house_dna
		cob_disputed_lands_cleanup
		cob_set_dragonstone_faiths
		cob_set_birthplaces
		cob_set_contracts
    }
}

# Bye map
lv_remove_realms = {

		lv_remove_realm_rhoyne

		lv_remove_realm_ghiscar
		lv_remove_realm_far_east_essos
		lv_remove_realm_sothoryos
    }
}

# Set up culture and history-related stuff here
lv_culture_and_history_setup = {
	effect = {
		# This prevents us from using direct file overrides if we don't have to and is generally cleaner for cases such as AGOT itself updating those cultures
		culture:mantaryan = {
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_mantaryan } }
				add_culture_tradition = lv_tradition_mantaryan
			}
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_north_valyrian_1 } }
				add_culture_tradition = lv_tradition_north_valyrian_1
			}
		}
		culture:tolosi = {
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_tolosi } }
				add_culture_tradition = lv_tradition_tolosi
			}
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_north_valyrian_1 } }
				add_culture_tradition = lv_tradition_north_valyrian_1
			}
		}
		culture:elyrian = {
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_elyrian } }
				add_culture_tradition = lv_tradition_elyrian
			}
			if = {
				limit = { NOT = { has_cultural_tradition = lv_tradition_north_valyrian_1 } }
				add_culture_tradition = lv_tradition_north_valyrian_1
			}
		}

		# Give Old Blood modifiers to Volantene dynasties
        dynasty:dynn_Maegyr = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
		dynasty:dynn_Vhassar = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Paenymion = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Qhaedar = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Vaelaros = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Staegone = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Tagaros = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }
        dynasty:dynn_Nogarys = { add_dynasty_modifier = valyrian_heritage_dynasty_modifier }

# Inconsequential vassal generator and title distributor
# - Will generate vassals for characters that exist in history that hold all titles in their respective duchies
# - Add entries here for duchies you don't really want to create an entire dedicated history for (mainly county-level vassals)
# - Parameters:
# - > GRANT_DUCHY - Whether to grant the holder of the county the duchy title on game start (you can either use this or just give them the duchy via history)
# - > DUCHY - The corresponding duchy title
# - > GIVE_COUNTY_CAPITAL - Whether to give away the capital county (or special cases such as Selhoru) to generated vassals as well
# - > COUNTY_OF_HOLDER - The "capital county" in reference
# - > GRANT_INDEPENDENCE - Whether to make the generated vassals break away on game start
# - > INHERIT_LIEGE_GOVERNMENT - Whether the new vassal inherits the liege's government type
# - > GOVERNMENT - Government to adopt if set to not inherit liege's government. If set to inherit, serves as the fallback government
# - > VASSAL_CULTURE - Culture of the new vassal
# - > VASSAL_FAITH - Faith of the new vassal
# - > VASSAL_FEMALE_CHANCE - How likely is the new vassal to be female
lv_vassals_setup = {
	effect = {
		every_ruler = {
			limit = {
				any_held_title = {
					OR = {
						# VOLANTIS
						AND = {
							lv_realm_volantis_disabled = no
							this = title:c_old_volantis
						}

						# ASSHAI
						AND = {
							lv_realm_far_east_essos_disabled = no
							OR = {
								this = title:c_asshai
							this = title:c_maruza
							this = title:c_ostiros
							}
						}
					}
				}
			}
			## VOLANTIS
			# VOLANTIS
			if = {
				limit = { this = title:c_old_volantis.holder }
				lv_distribute_titles_effect = {
					GRANT_DUCHY = no
					DUCHY = title:d_volantis
					GIVE_COUNTY_CAPITAL = no
					COUNTY_OF_HOLDER = title:c_old_volantis
					GRANT_INDEPENDENCE = no
					INHERIT_LIEGE_GOVERNMENT = yes
					GOVERNMENT = administrative_government
					VASSAL_CULTURE = this.culture
					VASSAL_FAITH = this.faith
					VASSAL_FEMALE_CHANCE = 50
				}
			}

			# ASSHAI
			if = {
				limit = { this = title:c_asshai.holder }
				lv_distribute_titles_effect = {
					GRANT_DUCHY = no
					DUCHY = title:d_asshai
					GIVE_COUNTY_CAPITAL = no
					COUNTY_OF_HOLDER = title:c_asshai
					GRANT_INDEPENDENCE = no
					INHERIT_LIEGE_GOVERNMENT = yes
					GOVERNMENT = feudal_government
					VASSAL_CULTURE = this.culture
					VASSAL_FAITH = this.faith
					VASSAL_FEMALE_CHANCE = 50
				}
			}
			if = {
				limit = { this = title:c_maruza.holder }
				lv_distribute_titles_effect = {
					GRANT_DUCHY = no
					DUCHY = title:d_maruza
					GIVE_COUNTY_CAPITAL = yes
					COUNTY_OF_HOLDER = title:c_maruza
					GRANT_INDEPENDENCE = no
					INHERIT_LIEGE_GOVERNMENT = yes
					GOVERNMENT = feudal_government
					VASSAL_CULTURE = this.culture
					VASSAL_FAITH = this.faith
					VASSAL_FEMALE_CHANCE = 50
				}
			}
			if = {
				limit = { this = title:c_ostiros.holder }
				lv_distribute_titles_effect = {
					GRANT_DUCHY = no
					DUCHY = title:d_ostiros
					GIVE_COUNTY_CAPITAL = yes
					COUNTY_OF_HOLDER = title:c_ostiros
					GRANT_INDEPENDENCE = no
					INHERIT_LIEGE_GOVERNMENT = yes
					GOVERNMENT = feudal_government
					VASSAL_CULTURE = this.culture
					VASSAL_FAITH = this.faith
					VASSAL_FEMALE_CHANCE = 50
				}
			}
		}
	}
}

lv_on_game_start = {
	effect = {
		set_global_variable = { # Set global variable for other submods to hook onto
            name = LV_is_loaded
            value = yes
        }

		# Yes!
		if = {
			limit = { global_var:LV_is_loaded ?= yes }
			debug_log = "Legacy of Valyria is loaded"
		}
	}
}
