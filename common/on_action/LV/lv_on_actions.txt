# Inconsequential vassal generator and title distributor
# - Will generate vassals for characters that exist in history that hold all titles in their respective duchies
# - Add entries here for duchies you don't really want to create an entire dedicated history for (mainly county-level vassals)
# - Parameters:
# - > GRANT_DUCHY - Whether to grant the holder of the county the duchy title on game start (you can either use this or just give them the duchy via history)
# - > DUCHY - The corresponding duchy title
# - > GIVE_COUNTY_CAPITAL - Whether to give away the capital county (or special cases such as Selhoru) to generated vassals as well
# - > COUNTY_OF_HOLDER - The "capital county" in reference
# - > GRANT_INDEPENDENCE - Whether to make the generated vassals break away on game start
# - > INHERIT_LIEGE_GOVERNMENT - Whether the new vassal inherits the liege's government type
# - > GOVERNMENT - Government to adopt if set to not inherit liege's government. If set to inherit, serves as the fallback government
# - > VASSAL_CULTURE - Culture of the new vassal
# - > VASSAL_FAITH - Faith of the new vassal
# - > VASSAL_FEMALE_CHANCE - How likely is the new vassal to be female

lv_on_game_start = {
	effect = {
		set_global_variable = { # Set global variable for other submods to hook onto
            name = LV_is_loaded
            value = yes
        }

		title:k_volantis = {
			if = {
				limit = { exists = holder }
				holder = {
					every_vassal_or_below = {
						limit = {
							highest_held_title_tier >= tier_county
							NOT = {
								OR = {
									this = title:d_the_red_priesthood.holder
									this = title:c_temple_of_the_lord_of_light.holder
								}
							}
						}
						if = {
							limit = { has_dlc_feature = admin_gov }
							change_government = administrative_government
						}
						else = { change_government = republic_government }
					}
				}
				set_variable = {
					name = years_since_election
					value = title_held_years
				}
			}
		}
		title:k_the_basilisk_isles = {
			if = {
				limit = { exists = holder }
				holder = {
					every_vassal_or_below = {
						limit = { highest_held_title_tier >= tier_county }
						if = {
							limit = { has_dlc_feature = admin_gov }
							change_government = administrative_government
						}
						else = { change_government = republic_government }
					}
				}
				set_variable = {
					name = years_since_election
					value = title_held_years
				}
			}
		}
		title:k_gogossos = {
			if = {
				limit = { exists = holder }

					every_vassal_or_below = {

		}
	}
}

on_game_start_after_lobby = {
	on_actions = {
		lv_ruins_setup
		lv_ruins_setup_non_default
		lv_ruins_setup_coa
	}
}

lv_ruins_setup = {
	trigger = { NOT = { has_game_rule = lv_colonisation_agot } }
	effect = {
		# Set up blockers
		every_county_in_region = {
			limit = {
				title_province = {
					OR = {
						has_holding_type = ruin_holding
						has_holding_type = wilderness_holding
					}
				}
			}
			region = world_valyria_mainland
			title_province = {
				# PREDEFINED BLOCKERS
				if = {
					limit = { this = province:9437 }
					lv_add_blocker_effect = {
						FORCE = yes
						BLOCKER_BUILDING = lokirian_barracks_01
					}
				}
				lv_blocker_generation_predefined_ruined_holdings_effect = yes
				lv_blocker_generation_predefined_mines_effect = yes
				lv_blocker_generation_predefined_noxious_gases_effect = yes

				# RANDOMLY-GENERATED BLOCKERS
				if = {
					limit = { NOT = { has_game_rule = lv_colonisation_lite } }
					lv_blocker_generation_random_noxious_gases_effect = yes
					lv_blocker_generation_random_unexplored_caves_effect = yes
					lv_blocker_generation_random_swamp_effect = yes
					lv_blocker_generation_random_stonemen_encampment_effect = yes
					lv_blocker_generation_random_bandit_camps_effect = yes
				}
			}
		}
	}
}

# Assign 1 blocker count with the idea of the holding itself being a blocker
# Basically: One upgraded settlement = reduce blocker count by 1
# Handy for game rules like Lite or AGOT, where there aren't any other blockers
lv_ruins_setup_non_default = {
	trigger = { NOT = { has_game_rule = lv_colonisation_default } }
	effect = {
		every_county_in_region = {
			limit = { title_province = { has_holding_type = ruin_holding } }
			region = world_valyria_mainland
			title_province = { lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = yes } }
		}
	}
}
