# This store cycle serves to facilitate firewyrm events
lv_firewyrm_story_cycle = {
	on_setup = {
		story_owner = {
			location.county ?= {
				if = {
					limit = { holder = { agot_is_dummy_character = yes } }
					scope:story = { end_story = yes }
				}
			}
			# Apply mood if set
			if = {
				limit = { exists = var:lv_firewyrm_mood }
				switch = {
					trigger = var:lv_firewyrm_mood
					flag:agitated = {
						scope:story = {
							set_variable = {
								name = lv_firewyrm_mood
								value = flag:agitated
							}
						}
					}
					flag:enraged = {
						scope:story = {
							set_variable = {
								name = lv_firewyrm_mood
								value = flag:enraged
							}
						}
					}
				}
				remove_variable = lv_firewyrm_mood
			}
			else = {
				scope:story = {
					set_variable = {
						name = lv_firewyrm_mood
						value = flag:dormant
					}
				}
			}
		}
	}

	on_owner_death = { end_story = yes }

	on_end = {
		story_owner = { save_scope_as = firewyrm }
		scope:firewyrm = {
			if = {
				limit = { is_alive = yes }
				death = { death_reason = death_vanished }
			}
			location = {
				save_scope_as = former_lair
				county ?= {
					holder = {
						if = { # Send notification to county owner
							limit = { agot_is_dummy_character = no }
							trigger_event = lv_firewyrm_event.0100
						}
					}
				}
			}
		}
	}

	# Daily
	effect_group = {
		days = 1
		triggered_effect = {
			trigger = { story_owner = { location.county ?= { holder = { agot_is_dummy_character = no } } } }
			effect = {
				# Rampage
				if = {
					limit = { NOT = { var:lv_firewyrm_mood ?= flag:dormant } }
					if = { # Rampage
						limit = {
							var:lv_firewyrm_mood ?= flag:enraged
							NOT = { has_variable = lv_firewyrm_rampage_countdown }
						}
						story_owner = {
							random_list = {
								90 = {}
								10 = {
									save_scope_as = firewyrm
									location = {
										save_scope_as = firewyrm_location
										county.holder = { trigger_event = lv_firewyrm_event.1000 }
									}
									scope:story = {
										set_variable = {
											name = lv_firewyrm_mood
											value = flag:dormant
										}
									}
								}
							}
						}
					}
				}
			}
		}
		fallback = { end_story = yes }
	}

	# Monthly
	effect_group = {
		days = 30
		triggered_effect = {
			trigger = { story_owner = { location.county ?= { holder = { agot_is_dummy_character = no } } } }
			effect = {
				# Update mood
				scope:story = {
					if = {
						limit = { var:lv_firewyrm_mood ?= flag:dormant }
						random_list = {
							98 = {}
							1 = { lv_firewyrm_change_mood_effect = { MOOD = agitated } } # Agitated
							1 = { lv_firewyrm_change_mood_effect = { MOOD = enraged } } # Enraged
						}
					}
					else_if = {
						limit = { var:lv_firewyrm_mood ?= flag:agitated }
						random_list = {
							80 = {}
							1 = { lv_firewyrm_change_mood_effect = { MOOD = dormant } } # Dormant
							19 = { lv_firewyrm_change_mood_effect = { MOOD = enraged } } # Enraged
						}
					}
				}
			}
		}
		fallback = { end_story = yes }
	}
}
