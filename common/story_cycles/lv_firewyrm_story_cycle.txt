# This store cycle serves to facilitate variable storage
lv_firewyrm_story_cycle = {
	on_setup = {
		story_owner = {
			location = {
				if = {
					limit = { has_free_building_slot = yes }
					scope:story = {
						set_variable = {
							name = lv_firewyrm_location
							value = prev
						}
					}
					set_variable = {
						name = lv_firewyrm
						value = prev
					}
					#lv_add_blocker_effect = {
					#	FORCE = no
					#	BLOCKER_BUILDING = firewyrm_lair
					#}
					add_building = firewyrm_lair
				}
				else_if = {
					limit = {
						county = {
							any_neighboring_county = {
								title_province = {
									NAND = {
										has_variable = lv_firewyrm
										has_building = firewyrm_lair
									}
									has_free_building_slot = yes
									save_temporary_scope_as = new_lair
								}
							}
						}
					}
					scope:new_lair = {
						scope:story = {
							set_variable = {
								name = lv_firewyrm_location
								value = prev
							}
						}
						set_variable = {
							name = lv_firewyrm
							value = prev
						}
						add_building = firewyrm_lair
					}
				}
				else = { scope:story = { end_story = yes } }
			}
		}
	}

	on_owner_death = { end_story = yes }

	on_end = {
		story_owner = { save_scope_as = firewyrm }
		var:lv_firewyrm_location = {
			remove_building = firewyrm_lair
			if = {
				limit = { has_variable = lv_firewyrm }
				remove_variable = lv_firewyrm
			}
			save_scope_as = former_lair
			county = {
				holder = {
					# Send notification to county owner
					trigger_event = lv_firewyrm_event.0100
				}
			}
		}
		remove_variable = lv_firewyrm_location
	}

	# Consider migrations and measures to keep them in the province if they have a lair if it ever becomes an issue eventually
	#effect_group = {
	#	days = 1
	#	trigger = { story_owner = { NOT = { var:lv_firewyrm_location ?= location } } }
	#	fallback = {
	#
	#	}
	#}

	# Stay at your lair bozo
	effect_group = {
		days = 1
		trigger = { story_owner = { NOT = { location ?= scope:story.var:lv_firewyrm_location } } }
		fallback = {
			var:lv_firewyrm_location = { save_scope_as = lair_location }
			story_owner = {
				set_employer = scope:lair_location.province_owner
				return_to_court = yes
			}
		}
	}

	# If firewyrm gets tamed, end the story and send a special event to province owner
	effect_group = {
		days = 1
		trigger = { story_owner = { has_character_flag = owned_dragon } }
		fallback = { end_story = yes }
	}
}
