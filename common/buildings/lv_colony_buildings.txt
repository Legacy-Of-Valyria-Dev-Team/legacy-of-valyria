colony_port_01 = {
	cost_gold = 100
	construction_time = 250
	type_icon = "icon_building_tradeport.dds"

	can_construct = {
		has_holding_type = settlement_holding
	}

	can_construct_potential = {
		building_common_tradeport_requirement_terrain = yes
		scope:holder = {
			NOR = {
				government_has_flag = government_is_clan
				government_has_flag = government_is_tribal
			}
		}
	}

	can_construct_showing_failures_only = {
		custom_description = {
			text = "must_not_have_blockers"
			NOT = { has_building_with_flag = obstacle }
		}
	}

	province_modifier = {
		monthly_income = normal_building_tax_tier_1
	}

	county_modifier = {
		development_growth = 0.05
	}

	character_modifier = {}

	type = regular
}
