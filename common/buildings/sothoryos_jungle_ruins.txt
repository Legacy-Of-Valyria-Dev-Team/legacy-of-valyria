# Jungle Ruin
jungle_ruin_01 = {
	can_construct_potential = {
		always = no
	}

	type_icon = "icon_building_ruin.dds"
	next_building = jungle_ruin_02
}

jungle_ruin_02 = {
	construction_time = 300

	can_construct_potential = {
		has_building = jungle_ruin_01
		custom_tooltip = {
			text = "RUIN_MUST_BEGIN_CONSTRUCTION"
			NOT = {
				scope:holder = {
					has_trait = ruin
				}
			}
		}
	}

	cost_gold = 150

	character_modifier = {
		monthly_income = -0.5
	}

	levy = 200
	max_garrison = 200
	garrison_reinforcement_factor = 0.05

	type_icon = "icon_building_ruin.dds"
	next_building = jungle_ruin_03
}

jungle_ruin_04 = {
	construction_time = 300

	can_construct_potential = {
		has_building = jungle_ruin_03
		custom_tooltip = { #LEGACY OF VALYRIA ADDED
			text = blockers_remain_in_settlement
			NOT = { has_building_with_flag = obstacle }
		}
		custom_tooltip = {
			text = "RUIN_MUST_BEGIN_CONSTRUCTION"
			NOT = {
				scope:holder = {
					has_trait = ruin
				}
			}
		}
	}

	on_complete = {
		save_scope_as = completed_ruin
		county.holder = { trigger_event = agot_ruins.1000 }
	}
	
	cost_gold = 250

	character_modifier = {
		monthly_income = -1.5
	}

	province_modifier = {
		fort_level = 3
	}

	levy = 400
	max_garrison = 400
	garrison_reinforcement_factor = 0.09

	type_icon = "icon_building_ruin.dds"
}
