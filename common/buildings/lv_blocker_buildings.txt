						value = flag:dragon_egg
					}
				}
				4 = { # Find a deposit of a random gem/metal
					set_variable = {
						name = blocker_outcome
						value = flag:gold_high
					}
				}
				4 = { # Find a deposit of a random gem/metal
					set_variable = {
						name = blocker_outcome
						value = flag:gold_low
					}
				}
				10 = {}
			}
			county.holder = { trigger_event = lv_blocker_clearance.0001 }
		}
	}

	flag = wilderness_blocker
	effect_desc = lv_wilderness_blocker_effect
}
