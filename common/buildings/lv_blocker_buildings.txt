unexplored_caves_03 = {
	cost_gold = 200
	construction_time = 720
	type_icon = "icon_structure_vatapi_caves.dds"

	can_construct_potential = { scope:holder = { is_human = yes } }

	on_complete = {
		save_scope_as = blocker_province
		hidden_effect = {
			county.holder = {
				random_list = {
					1 = { # find a dragon egg
						trigger_event = blockers.0002
					}
					4 = { # Cave explored, deposit of a random gem/metal
						send_interface_toast = {
							title = find_gold_cache
							left_icon = this
							add_gold = 200
						}
					}
					4 = { # Cave explored, deposit of a random gem/metal
						send_interface_toast = {
							title = find_gold_cache
							left_icon = this
							add_gold = 100
						}
					}
					10 = {} # Clears the building as normal
				}
			}
			remove_building = unexplored_caves_03
		}
	}

	flag = wilderness_blocker
	effect_desc = lv_wilderness_blocker_effect
}
