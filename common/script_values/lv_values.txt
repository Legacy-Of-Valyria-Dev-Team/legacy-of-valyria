zz_valyria_max_wards = {
	value = 2
	if = {
		limit = {
			learning > 10
		}
		add = {
			value = learning
			divide = 10
		}
	}
}

lv_wasteland_valyria_location = {
	value = 0
	if = {
		limit = { any_county_province = { lv_is_fourteen_flames = yes } }
		value = 1
	}
	else_if = {
		limit = { any_neighboring_county = { any_county_province = { lv_is_fourteen_flames = yes } } }
		value = 0.75
	}
	else_if = {
		limit = { any_neighboring_county = { any_neighboring_county = { any_county_province = { lv_is_fourteen_flames = yes } } } }
		value = 0.5
	}
	else = { value = 0.25 }
}

lv_wasteland_valyria_progress = {
	value = 1
	multiply = {
		value = 1
		title_province = {
			multiply = {
				if = {
					limit = { has_variable = lv_blocker_count }
					value = var:lv_blocker_count
				}
				else = { value = 1 }
			}
			divide = {
				if = {
					limit = { has_variable = lv_blocker_total }
					value = var:lv_blocker_total
				}
				else = { value = 1 }
			}
		}
	}
}
