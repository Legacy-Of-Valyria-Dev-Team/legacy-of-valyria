# Legacy of Valyria firewyrm scripted effects
# - Mainly derived from AGOT's dragon combat system

lv_spawn_firewyrm_effect = {
	create_character = {
		template = lv_firewyrm_template
		save_scope_as = firewyrm
		location = $LOCATION$
		after_creation = { add_character_flag = dragon_idle }
	}
	if = {
		limit = { always = $GENERATE_BLOCKER$ }
		scope:firewyrm = {
			create_story = {
				type = lv_firewyrm_story_cycle
				save_scope_as = firewyrm_story
			}
			custom_tooltip = lv_firewyrm_spawned_tooltip
		}
	}
}

lv_apply_firewyrm_aging_effect = {
	set_variable = {
		name = dragon_age_counter
		value = 0
	}
	while = {
		limit = { age > var:dragon_age_counter }
		# Normal growth calculations
		if = {
			limit = { var:dragon_age_counter < 11 }
			add_prowess_skill = 4
			change_dragon_size_direct = { VALUE = 4 }
		}
		else_if = {
			limit = {
				var:dragon_age_counter < 15
				NOT = { has_trait = dragon_destined }
			}
			add_prowess_skill = 3
			change_dragon_size_direct = { VALUE = 3 }
		}
		else = {
			# Boo the location isn't set yet when creating a character smh
			#if = {
			#	limit = { location = { lv_is_fourteen_flames = yes } }
			#	add_prowess_skill = 2
			#	change_dragon_size_direct = { VALUE = 2 }
			#}
			#else = {
			#	add_prowess_skill = 1
			#	change_dragon_size_direct = {
			#		VALUE = 1
			#	}
			#}
			random_list = {
				50 = {
					add_prowess_skill = 1
					change_dragon_size_direct = { VALUE = 1 }
				}
				50 = {
					add_prowess_skill = 2
					change_dragon_size_direct = { VALUE = 2 }
				}
			}
		}
		change_variable = {
			name = dragon_age_counter
			add = 1
		}
	}
	remove_variable = dragon_age_counter
}

lv_slay_firewyrm_effect = {
	$SLAYER$ = { save_scope_as = human }
	$FIREWYRM$ = { save_scope_as = dragon }

	scope:dragon.location = { save_scope_as = background_wilderness_scope }

	# To prevent multiple dragon slaying attempts at the same time from player
	# and for dragon
	scope:human = { add_character_flag = in_dragon_combat }
	scope:dragon = { add_character_flag = in_dragon_combat }
	# Will be removed during events as needed
	if = {
		limit = {
			$SLAYER$ = {
				any_memory = {
					OR = {
						has_memory_type = spouse_died
						has_memory_type = soulmate_died
						has_memory_type = lover_died
						has_memory_type = best_friend_died
						has_memory_type = friend_died
						has_memory_type = relative_died
					}
					memory_participant:dead_relation ?= {
						trigger_if = {
							limit = { exists = killer }
							killer = $FIREWYRM$
						}
						trigger_else = { always = no }
						save_temporary_scope_as = dragon_kill
					}
				}
			}
		}
		scope:dragon_kill = { save_scope_as = vengeance_candidate }
	}
	agot_configure_start_dragon_slay_effect = {
		INITIATOR = scope:firewyrm
		HUMAN = scope:human
		DRAGON = scope:firewyrm
		PROXIMITY = 2
		LOCALE = wilderness_scope # Should be a field or whatever
		INVALIDATION_EVENT = single_combat.1006
	}
}

lv_start_firewyrm_duel_combat_effect = {
	$ATTACKER$ = {
		save_scope_as = fd_attacker
		var:current_dragon ?= { save_scope_as = fd_attacker_dragon }
	}
	$DEFENDER$ = {
		save_scope_as = fd_defender
		if = {
			limit = { has_trait = dragon }
			save_scope_as = fd_defender_dragon
		}
		else = { var:current_dragon ?= { save_scope_as = fd_defender_dragon } }
	}
	hidden_effect = { lv_remove_firewyrm_duel_combat_info_effect = yes }
	scope:fd_attacker = {
		set_variable = { # No simultanious vanilla combat
			name = engaged_in_single_combat
			value = yes
		}
		set_variable = { # Or simultanious dragon combat
			name = engaged_in_dragon_combat
			value = yes
		}
		set_variable = {
			name = fd_chance_of_winning
			value = 50
		}
	}
	scope:fd_defender = {
		set_variable = { # No simultanious vanilla combat
			name = engaged_in_single_combat
			value = yes
		}
		set_variable = { # Or simultanious dragon combat
			name = engaged_in_dragon_combat
			value = yes
		}
		set_variable = {
			name = fd_chance_of_winning
			value = 50
		}
		set_variable = {
			name = fd_combat_current_round
			value = 1
		}
	}
	trigger_event = lv_firewyrm_duel.0001
}

lv_remove_firewyrm_duel_combat_info_effect = {
	scope:fd_attacker = {
		if = {
			limit = { is_alive = yes }
			remove_firewyrm_duel_combat_variables_effect = yes
		}
	}
	scope:fd_attacker_dragon = {
		if = {
			limit = { is_alive = yes }
			remove_firewyrm_duel_combat_variables_effect = yes
		}
	}
	scope:fd_defender = {
		if = {
			limit = { is_alive = yes }
			remove_firewyrm_duel_combat_variables_effect = yes
		}
	}
	scope:fd_defender_dragon = {
		if = {
			limit = { is_alive = yes }
			remove_firewyrm_duel_combat_variables_effect = yes
		}
	}
}

remove_firewyrm_duel_combat_variables_effect = {
	clear_variable_list = fd_combat_moves

	#remove_variable = fd_tier_1_counter
	#remove_variable = fd_tier_2_counter
	#remove_variable = fd_tier_3_counter
	remove_variable = fd_chance_of_winning
	remove_variable = fd_combat_current_round
	# Note, we don't remove var:engaged_in_single_combat or engaged_in_dragon_combat here, as we run this effect at the start of combat for clean-up also.
}

# New Round Effects
lv_firewyrm_duel_post_move_effect = {
	if = {
		limit = { always = $START_FLIGHT$ }
		scope:fd_round_me_dragon = {
			set_variable = {
				name = lv_dragon_state
				value = flag:flying
			}
		}
	}
	if = {
		limit = { always = $FORCE_LAND$ }
		scope:fd_round_me_dragon = {
			if = {
				limit = { var:lv_dragon_state ?= flag:flying }
				custom_tooltip = {
					text = lv_firewyrm_duel.0001.warning.force_land
					remove_variable = lv_dragon_state
				}
			}
		}
	}
	if = {
		limit = {
			always = $INVOLVES_CONTACT$
			scope:fd_round_opponent_dragon = { var:lv_dragon_state ?= flag:molten }
		}
		custom_tooltip = {
			text = lv_firewyrm_duel.0001.warning.burn_contact
			scope:fd_round_me_dragon = {
				if = {
					limit = { has_trait = dragon_burned }
					wound_dragon = yes
				}
				else = { add_trait = dragon_burned }
			}
		}
	}
	if = {
		limit = {
			NOR = {
				always = $START_FLIGHT$
				scope:fd_round_me_dragon = {
					OR = {
						var:lv_dragon_state ?= flag:flying
						var:lv_dragon_state ?= flag:molten
					}
				}
			}
			scope:fd_round_opponent_dragon = { var:lv_dragon_state ?= flag:cone_of_flame }
		}
		custom_tooltip = {
			text = lv_firewyrm_duel.0001.warning.burn_incoming
			scope:fd_round_me_dragon = {
				if = {
					limit = { has_trait = dragon_burned }
					wound_dragon = yes
				}
				else = { add_trait = dragon_burned }
			}
		}
	}
}

# Finalize results
finalise_firewyrm_duel_combat_results_effect = {
	# Determine what death rattle is appropriate
	scope:fd_loser = {
		if = {
			limit = { this = scope:fd_attacker }
			var:fd_last_move = { save_scope_as = fd_victors_last_move }
		}
		else = { var:fd_last_move = { save_scope_as = fd_victors_last_move } }
		scope:fd_victors_last_move = {
			if = {
				limit = {
					OR = {
						this = flag:dragon_claws_out
						this = flag:dragon_vipers_feint
						this = flag:dragon_rip_joints
						this = flag:dragon_hidden_knowledge
						this = flag:dragon_commanding_cry
						this = flag:firewyrm_boneshattering_screech
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:sudden_strike
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_flame_test
						this = flag:dragon_dragonflame_barrage
						this = flag:dragon_hidden_knowledge
						this = flag:firewyrm_fume_burst
						this = flag:firewyrm_cone_of_flame
						this = flag:firewyrm_molten_slough
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:dragonfire
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_tailspin_slam
						this = flag:dragon_dive_strike
						this = flag:dragon_collapse_ground
						#AND = {
						#	this = flag:firewyrm_frenzied_lunge
						#	ROOT = {
						#		location = {
						#			OR = {
						#				# hazardous terrain types here
						#			}
						#		}
						#	}
						#}
						this = flag:firewyrm_rockfall_roar
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:driven_to_ground
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_reckless_dive
						this = flag:dragon_rip_joints
						this = flag:dragon_wound_exploit
						this = flag:firewyrm_frenzied_lunge
						this = flag:firewyrm_berserk_charge
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:disemboweled
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_enthusiastic_onslaught
						this = flag:dragon_onslaught
						this = flag:dragon_expert_onslaught
						this = flag:dragon_lightning_assault
						this = flag:dragon_martial_voice
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:eviscerated
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_chomp
						this = flag:dragon_risk_it_all
						this = flag:firewyrm_chomp
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:chomp
				}
			}
			# Unique to Firewyrm: Suffocated
			# Unique to Firewyrm: Dragged into the Deep
		}
	}
	scope:fd_loser = {
		if = {
			limit = { has_trait = dragon }
			save_scope_as = fd_loser_dragon
		}
		else = { var:current_dragon ?= { save_scope_as = fd_loser_dragon } }
		trigger_event = lv_firewyrm_duel.0003
	}
	scope:fd_victor = {
		if = {
			limit = { has_trait = dragon }
			save_scope_as = fd_victor_dragon
		}
		else = { var:current_dragon ?= { save_scope_as = fd_victor_dragon } }
		trigger_event = lv_firewyrm_duel.0004
	}
	scope:fd_victor = {
		if = {
			limit = { exists = accolade }
			# Glory is relative to the prowess difference between two characters. Beating someone equally matched or better is worth more.
			## Massively outclassed.
			if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -20
					}
				}
				accolade = { add_glory = monumental_glory_gain }
			}
			## Majorly outclassed.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -15
					}
				}
				accolade = { add_glory = massive_glory_gain }
			}
			## Somewhat outclassed.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -8
					}
				}
				accolade = { add_glory = major_glory_gain }
			}
			## Slightly outclassed.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -4
					}
				}
				accolade = { add_glory = medium_glory_gain }
			}
			## Evenly matched.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= 0
					}
				}
				accolade = { add_glory = minor_glory_gain }
			}
			## Slightly in your favour.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= 4
					}
				}
				accolade = { add_glory = minimal_glory_gain }
			}
		}
	}
	lv_firewyrm_duel_combat_result_grab = yes
	lv_remove_firewyrm_duel_combat_info_effect = yes
	scope:fd_defender = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
			remove_variable = engaged_in_dragon_combat
		}
	}
	scope:fd_attacker = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
			remove_variable = engaged_in_dragon_combat
		}
	}
}
