# Legacy of Valyria firewyrm scripted effects
# - Mainly derived from AGOT's dragon combat system

lv_spawn_firewyrm_effect = {
	create_character = {
		template = lv_firewyrm_template
		save_scope_as = firewyrm
		location = $LOCATION$
		after_creation = { add_character_flag = dragon_idle }
	}
	if = {
		limit = { always = $GENERATE_BLOCKER$ }
		scope:firewyrm = {
			create_story = {
				type = lv_firewyrm_story_cycle
				save_scope_as = firewyrm_story
			}
			custom_tooltip = lv_firewyrm_spawned_tooltip
		}
	}
}

lv_apply_firewyrm_aging_effect = {
	set_variable = {
		name = dragon_age_counter
		value = 0
	}
	while = {
		limit = { age > var:dragon_age_counter }
		# Normal growth calculations
		if = {
			limit = { var:dragon_age_counter < 11 }
			add_prowess_skill = 4
			change_dragon_size_direct = { VALUE = 4 }
		}
		else_if = {
			limit = {
				var:dragon_age_counter < 15
				NOT = { has_trait = dragon_destined }
			}
			add_prowess_skill = 3
			change_dragon_size_direct = { VALUE = 3 }
		}
		else = {
			# Boo the location isn't set yet when creating a character smh
			#if = {
			#	limit = { location = { lv_is_fourteen_flames = yes } }
			#	add_prowess_skill = 2
			#	change_dragon_size_direct = { VALUE = 2 }
			#}
			#else = {
			#	add_prowess_skill = 1
			#	change_dragon_size_direct = {
			#		VALUE = 1
			#	}
			#}
			random_list = {
				50 = {
					add_prowess_skill = 1
					change_dragon_size_direct = { VALUE = 1 }
				}
				50 = {
					add_prowess_skill = 2
					change_dragon_size_direct = { VALUE = 2 }
				}
			}
		}
		change_variable = {
			name = dragon_age_counter
			add = 1
		}
	}
	remove_variable = dragon_age_counter
}

lv_firewyrm_death_tooltip_effect = {
	scope:firewyrm = {
		if = {
			limit = {
				any_owned_story = {
					story_type = lv_firewyrm_story_cycle
					var:lv_firewyrm_location ?= { save_temporary_scope_as = lair_location }
				}
			}
			custom_tooltip = lv_firewyrm_death_tooltip
		}
	}
}

lv_start_firewyrm_vs_dragon_combat_effect = {
	$DRAGONRIDER$ = {
		save_scope_as = fd_dragonrider
		var:current_dragon ?= { save_scope_as = fd_dragon }
	}
	$FIREWYRM$ = { save_scope_as = fd_firewyrm }
	hidden_effect = { lv_remove_firewyrm_vs_dragon_combat_info_effect = yes }
	scope:fd_dragonrider = {
		set_variable = { # No simultanious vanilla combat
			name = engaged_in_single_combat
			value = yes
		}
		set_variable = { # Or simultanious dragon combat
			name = engaged_in_dragon_combat
			value = yes
		}
		set_variable = {
			name = fd_chance_of_winning
			value = 50
		}
		set_variable = {
			name = fd_combat_current_round
			value = 1
		}
	}
	# We don't *really* need this since the Firewyrm gets purged anyway, but just for good measure
	scope:fd_firewyrm = {
		set_variable = { # No simultanious vanilla combat
			name = engaged_in_single_combat
			value = yes
		}
		set_variable = { # Or simultanious dragon combat
			name = engaged_in_dragon_combat
			value = yes
		}
		set_variable = {
			name = fd_chance_of_winning
			value = 50
		}
	}
	trigger_event = lv_firewyrm_duel.0001
}

lv_remove_firewyrm_vs_dragon_combat_info_effect = {
	scope:fd_dragonrider = {
		if = {
			limit = { is_alive = yes }
			remove_firewyrm_vs_dragon_combat_variables_effect = yes
		}
	}
	scope:fd_dragon = {
		if = {
			limit = { is_alive = yes }
			remove_firewyrm_vs_dragon_combat_variables_effect = yes
		}
	}
	scope:fd_firewyrm = {
		if = {
			limit = { is_alive = yes }
			remove_variable = fd_chance_of_winning
		}
	}
}

remove_firewyrm_vs_dragon_combat_variables_effect = {
	clear_variable_list = fd_combat_moves

	#remove_variable = fd_tier_1_counter
	#remove_variable = fd_tier_2_counter
	#remove_variable = fd_tier_3_counter
	remove_variable = fd_chance_of_winning
	remove_variable = fd_combat_current_round
	# Note, we don't remove var:engaged_in_single_combat or engaged_in_dragon_combat here, as we run this effect at the start of combat for clean-up also.
}

lv_select_special_firewyrm_vs_dragon_tier_move_effect = {
	lv_add_firewyrm_vs_dragon_move_to_list_effect = { FLAG = dragon_special_slot_used_flag }
	random_list = {
		# CHOMP - insta kill
		9999 = {
			trigger = {
				NOT = {
					any_in_list = {
						variable = fd_combat_moves
						this = flag:dragon_chomp
					}
				}
				save_temporary_scope_value_as = {
					name = firewyrm_size_difference
					value = {
						value = 0
						add = scope:fd_dragon.dragon_combat_effectiveness
						subtract = scope:fd_firewyrm.dragon_combat_effectiveness
					}
				}
				# Dragon must be at least twice the size of the firewyrm (scary!)
				scope:firewyrm_size_difference > scope:fd_firewyrm.dragon_combat_effectiveness
			}
			lv_add_firewyrm_vs_dragon_move_to_list_effect = { FLAG = dragon_chomp }
		}
		# Commanding Cry
		1000 = {
			trigger = {
				NOT = {
					any_in_list = {
						variable = fd_combat_moves
						this = flag:dragon_commanding_cry
					}
				}
				diplomacy >= very_high_skill_rating
				NAND = {
					this = scope:fd_dragonrider
					scope:fd_dragon_last_move ?= flag:dragon_commanding_cry
				}
				this = scope:fd_dragonrider
				scope:fd_dragon = {
					OR = {
						has_trait = dragon_wounded_1
						has_trait = dragon_wounded_2
						has_trait = dragon_wounded_3
						has_trait = dragon_wounded_4
						has_trait = dragon_wounded_5
					}
				}
			}
			lv_add_firewyrm_vs_dragon_move_to_list_effect = { FLAG = dragon_commanding_cry }
		}
		# Viper's Feint
		1000 = {
			trigger = {
				var:lv_dragon_state ?= flag:flying
				NOT = {
					any_in_list = {
						variable = fd_combat_moves
						this = flag:dragon_vipers_feint
					}
				}
				intrigue >= very_high_skill_rating
				NAND = {
					this = scope:fd_dragonrider

		# Hidden Knowledge

# New Round Effects
lv_firewyrm_vs_dragon_post_move_effect = {
	if = {
		limit = { always = $START_FLIGHT$ }
		set_variable = {
			name = lv_dragon_state
			value = flag:flying
		}
	}
	if = {
		limit = { always = $FORCE_LAND$ }
		if = {
			limit = { var:lv_dragon_state ?= flag:flying }
			custom_tooltip = {
				text = lv_firewyrm_duel.0001.warning.force_land
				remove_variable = lv_dragon_state
			}
		}
	}
	if = {
		limit = {
			always = $INVOLVES_CONTACT$
			scope:fd_firewyrm = { var:lv_firewyrm_state ?= flag:molten }
		}
		custom_tooltip = {
			text = lv_firewyrm_duel.0001.warning.burn_contact
			scope:fd_dragon = {
				if = {
					limit = { has_trait = dragon_burned }
					wound_dragon = yes
				}
				else = { add_trait = dragon_burned }
			}
		}
	}
	if = {
		limit = {
			NOR = {
				always = $START_FLIGHT$
				AND = { var:lv_dragon_state ?= flag:flying }
			}
			scope:fd_firewyrm = { var:lv_firewyrm_state ?= flag:cone_of_flame }
		}
		custom_tooltip = {
			text = lv_firewyrm_duel.0001.warning.burn_incoming
			scope:fd_dragon = {
				if = {
					limit = { has_trait = dragon_burned }
					wound_dragon = yes
				}
				else = { add_trait = dragon_burned }
			}
		}
	}
}

# Finalize results
finalise_firewyrm_vs_dragon_combat_results_effect = {
	# Determine what death rattle is appropriate
	scope:fd_loser = {
		if = {
			limit = { this = scope:fd_dragonrider }
			scope:fd_firewyrm_last_move = { save_scope_as = fd_victors_last_move }
		}
		else = { scope:fd_dragon_last_move = { save_scope_as = fd_victors_last_move } }
		scope:fd_victors_last_move = {
			if = {
				limit = {
					OR = {
						this = flag:dragon_vipers_feint
						this = flag:dragon_rip_joints
						this = flag:dragon_hidden_knowledge
						this = flag:firewyrm_molten_slough
						this = flag:dragon_commanding_cry
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:sudden_strike
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_flame_test
						this = flag:dragon_dragonflame_barrage
						this = flag:dragon_hidden_knowledge
						this = flag:firewyrm_cone_of_flame
						this = flag:firewyrm_fume_burst
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:dragonfire
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_tailspin_slam
						this = flag:dragon_dive_strike
						this = flag:dragon_collapse_ground
						#AND = {
						#	this = flag:firewyrm_frenzied_lunge
						#	ROOT = {
						#		location = {
						#			OR = {
						#				# hazardous terrain types here
						#			}
						#		}
						#	}
						#}
						this = flag:firewyrm_rockfall_roar
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:driven_to_ground
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_reckless_dive
						this = flag:dragon_rip_joints
						this = flag:dragon_wound_exploit
						this = flag:firewyrm_frenzied_lunge
						this = flag:firewyrm_berserk_charge
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:disemboweled
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_enthusiastic_onslaught
						this = flag:dragon_onslaught
						this = flag:dragon_expert_onslaught
						this = flag:dragon_lightning_assault
						this = flag:dragon_martial_voice
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:eviscerated
				}
			}
			else_if = {
				limit = {
					OR = {
						this = flag:dragon_chomp
						this = flag:dragon_risk_it_all
						this = flag:firewyrm_chomp
					}
				}
				save_scope_value_as = {
					name = dragon_death_rattle
					value = flag:chomp
				}
			}
			# Unique to Firewyrm: Suffocated
			# Unique to Firewyrm: Dragged into the Deep
		}
	}
	if = {
		limit = { this = scope:fd_loser }
		if = {
			limit = { this = scope:fd_dragonrider }
			trigger_event = lv_firewyrm_duel.0003
		}
		else = {
			trigger_event = lv_firewyrm_duel.0003
		}
	}
	else_if = {
		limit = { this = scope:fd_victor }
		trigger_event = lv_firewyrm_duel.0004
	}
	scope:fd_victor = {
		if = {
			limit = { accolade ?= { is_accolade_active = yes } }
			# Glory is relative to the prowess difference between two characters. Beating someone equally matched or better is worth more.
			## Massively outclassed.
			if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -20
					}
				}
				accolade = { add_glory = monumental_glory_gain }
			}
			## Majorly outclassed.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -15
					}
				}
				accolade = { add_glory = massive_glory_gain }
			}
			## Somewhat outclassed.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -8
					}
				}
				accolade = { add_glory = major_glory_gain }
			}
			## Slightly outclassed.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= -4
					}
				}
				accolade = { add_glory = medium_glory_gain }
			}
			## Evenly matched.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= 0
					}
				}
				accolade = { add_glory = minor_glory_gain }
			}
			## Slightly in your favour.
			else_if = {
				limit = {
					prowess_diff = {
						target = scope:fd_loser
						value <= 4
					}
				}
				accolade = { add_glory = minimal_glory_gain }
			}
		}
	}
	lv_firewyrm_vs_dragon_combat_result_grab = yes
	lv_remove_firewyrm_vs_dragon_combat_info_effect = yes
	scope:fd_firewyrm = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
			remove_variable = engaged_in_dragon_combat
		}
	}
	scope:fd_dragonrider = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
			remove_variable = engaged_in_dragon_combat
		}
	}
}
