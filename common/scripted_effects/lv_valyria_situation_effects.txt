# Legacy of Valyria situation scripted effects

# Requires: 'province' scope
# Output Scope: 'sub_region' scope
lv_valyria_retrieve_subregion_effect = {
	situation:valyria_situation ?= {
		every_situation_sub_region = {
			every_situation_sub_region_geographical_region = {
				limit = { scope:province ?= { geographical_region = prev } }
				save_scope_as = sub_region
			}
		}
	}
}

lv_valyria_progress_subregion_rehabilitation = {
	situation:valyria_situation = {
		situation_sub_region:$SUBREGION$ = {
			if = {
				limit = { sub_region_current_phase = valyria_ruined }
				change_phase = { phase = valyria_restored_01 }
			}
			else_if = {
				limit = { sub_region_current_phase = valyria_restored_01 }
				change_phase = { phase = valyria_restored_02 }
			}
			else_if = {
				limit = { sub_region_current_phase = valyria_restored_02 }
				change_phase = { phase = valyria_rehabilitated }
			}
		}
	}
}
