agot_create_spider_ambush = {
	create_character = {
		name = cos_name_spider_man
		gender = male
		template = crannogman_template
		save_scope_as = crannogman
		location = scope:target_location
		culture = culture:brindled_men
		faith = faith:eldritch_deep_dreams
	}
	# Set up title so that war can be declared
	create_dynamic_title = {
		tier = duchy
		name = cos_name_spider_duchy_title
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:new_title = {
		set_capital_county = $TARGET$.capital_county
		set_landless_title = yes
		set_destroy_on_succession = yes
		set_delete_on_destroy = yes
		set_no_automatic_claims = yes
		set_definitive_form = yes
		set_can_be_named_after_dynasty = no
		change_title_holder = {
			holder = scope:crannogman
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
}
agot_spawn_spider_army = { # TARGET
	scope:crannogman = {
		hidden_effect = { add_gold = 100 }
		start_war = {
			cb = cos_spider_ambush_cb
			target = $TARGET$
			target_title = $TARGET$.primary_title
		}
		if = {
			limit = {exists = scope:target_location }
			spawn_army = {
				name = spider_army_name
				levies = 0
				men_at_arms = {
					type = spotted_spider_matriarch
					stacks = 1
				}
				men_at_arms = {
					type = giant_spotted_spider
					stacks = 6
				}
				location = scope:target_location
			}
		}
		else = {
			spawn_army = {
				name = spider_army_name
				levies = 0
				men_at_arms = {
					type = spotted_spider_matriarch
					stacks = 1
				}
				men_at_arms = {
					type = giant_spotted_spider
					stacks = 6
				}
				location = $TARGET$.location
			}
		}
	}
}
