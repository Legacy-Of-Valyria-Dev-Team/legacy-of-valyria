# AGOT OVERWRITE
# Aurion Varezys
agot_init_Varezys_1 = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_character_flag = is_Varezys_1 }
		}
		add_character_flag = is_Varezys_1
	}
	if = {
		limit = { is_alive = yes }
		add_character_flag = hair_trait_male_valyrian_loose_hair_02
		add_character_flag = beard_trait_male_beard_fp2_iberian_christian_03
		make_trait_inactive = cool_silver_hair
	}
	else = {
		make_trait_inactive = hair_trait_male_valyrian_loose_hair_02
		make_trait_inactive = beard_trait_male_beard_fp2_iberian_christian_03
		make_trait_inactive = cool_silver_hair
	}
}
