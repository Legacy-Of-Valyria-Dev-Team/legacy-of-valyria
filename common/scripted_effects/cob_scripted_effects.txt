cob_tommen_artifact_yeet_effect = {
    if = {
        limit = { any_artifact = { has_artifact_modifier = vs_brightroar_modifier } }
        random_artifact = {
            limit = { has_artifact_modifier = vs_brightroar_modifier }
            save_scope_as = brightroar
        }
        destroy_artifact = scope:brightroar
        remove_list_global_variable = {
            name = unavailable_unique_decisions
            target = flag:recovered_brightroar
        }
    }
    if = {
        limit = { any_artifact = { has_artifact_modifier = westerlands2_crown_modifier } }
        random_artifact = {
            limit = { has_artifact_modifier = westerlands2_crown_modifier }
            save_scope_as = lion_crown
        }
        destroy_artifact = scope:lion_crown
    }
    if = {
        limit = {
            any_artifact = {
                NOT = {
                    has_artifact_modifier = westerlands2_crown_modifier
                    has_artifact_modifier = vs_brightroar_modifier
                }
            }
        }
        destroy_artifact = this
    }
}

cob_dragonlord_government_setup_effect = {
    every_in_de_jure_hierarchy = {
        set_variable = historical_slavery
        holder ?= {
            if = {
                limit = { NOT = { has_realm_law = slavery_allowed_law } }
                add_realm_law_skip_effects = slavery_allowed_law
            }
            if = {
                limit = {
                    has_realm_law = equal_law
                }
                remove_realm_law = equal_law
                add_realm_law = male_preference_law
            }
        }
    }
}

cob_riverlands_tributary_effect = {
    if = {
        limit = {
            game_start_date = 7899.8.14 # 7899.6.17 Temp change for AGOT history of Doom
        }
        title:e_the_riverlands.holder = {
            save_scope_as = suzerain
            title:e_the_riverlands = {
                every_dejure_vassal_title_holder = {
                    limit = {
                        is_vassal_of = scope:suzerain
                    }
                    break_subject_contract_and_establish_tributary_effect = { SUZERAIN = scope:suzerain TRIBUTARY = this }
                }
            }
            title:e_the_reach = {
                every_dejure_vassal_title_holder = {
                    limit = {
                        is_vassal_of = scope:suzerain
                    }
                    break_subject_contract_and_establish_tributary_effect = { SUZERAIN = scope:suzerain TRIBUTARY = this }
                }
            }
            title:e_the_crownlands = {
                every_dejure_vassal_title_holder = {
                    limit = {
                        is_vassal_of = scope:suzerain
                    }
                    break_subject_contract_and_establish_tributary_effect = { SUZERAIN = scope:suzerain TRIBUTARY = this }
                }
            }
        }
        title:c_ramsford.holder = {
            get_title = title:d_fairmarket
            get_title = title:c_fairmarket
            title:d_fairmarket = {
                every_dejure_vassal_title_holder = {
                    save_scope_as = vassal_to_be
                    create_title_and_vassal_change = {
                        type = swear_fealty
                        save_scope_as = change_fealty
                    }
                    scope:vassal_to_be = {
                        change_liege = {
                            liege = title:d_fairmarket.holder
                            change = scope:change_fealty
                        }
                    }
                    resolve_title_and_vassal_change = scope:change_fealty
                }
            }
        }
        title:e_the_riverlands.holder = {
            destroy_title = title:e_the_riverlands
        }
    }
}
