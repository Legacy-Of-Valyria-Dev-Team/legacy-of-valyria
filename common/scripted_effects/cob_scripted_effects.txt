cob_aurion_artifact_yeet_effect = {
    if = { #Aurion Armor Lost
        limit = {
            character:Varezys_1 = {
                var:cob_restore_valyria = flag:positive
            }
            any_artifact = { has_artifact_modifier = aurion_valyrian_steel_armor_modifier }
        }
        random_artifact = {
            limit = { has_artifact_modifier = aurion_valyrian_steel_armor_modifier }
            save_scope_as = aurion_armor
        }
        add_to_global_variable_list = {
            name = lv_unavailable_unique_items
            target = flag:aurion_armor_lost
        }
        destroy_artifact = scope:aurion_armor
    }
    if = { #Aurion Sword Lost
        limit = {
            character:Varezys_1 = {
                var:cob_restore_valyria = flag:positive
            }
            any_artifact = { has_artifact_modifier = vs_stormbringer_modifier }
        }
        random_artifact = {
            limit = { has_artifact_modifier = vs_stormbringer_modifier }
            save_scope_as = aurion_sword
        }
        add_to_global_variable_list = {
            name = lv_unavailable_unique_items
            target = flag:aurion_sword_lost
        }
        destroy_artifact = scope:aurion_sword
    }
}

cob_dynasty_banner_effect = {
    root = {
        set_variable = banner_from_dynasty
        create_artifact_wall_banner_effect = {
            OWNER = root
            CREATOR = root
            TARGET = root.dynasty
        }
    }
    scope:newly_created_artifact = {
        set_variable = { # Tracks house
            name = banner_dynasty
            value = scope:target
        }
    }
}
