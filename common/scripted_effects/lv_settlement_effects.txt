# Legacy of Valyria settlement scripted effects

lv_settle_destination_effect = {
	# Determine ruin level for converting to appropriate colony building
	$PROVINCE$ = {
		county = { save_scope_as = county }
		# Uncomment if considering using AGOT's default ruins system
		#if = {
		#	limit = { has_holding_type = wilderness_holding }
			set_holding_type = settlement_holding
		#}
	}
	$ACTOR$ = {
		# Liege "gets" the title first, so it will reflect in history that they "granted" the title to the settler
		if = {
			limit = { NOT = { exists = scope:liege } }
			save_scope_as = liege
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = claim_change
		}
		scope:county = {
			change_title_holder = {
				holder = scope:liege
				change = scope:claim_change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:claim_change

		# Now we give it to the actual settler
		if = {
			limit = {
				NOT = { this = scope:liege }
				scope:liege = {
					NOT = {
						# Can't have county vassals as a count, of course
						highest_held_title_tier = tier_county
					}
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
			}
			change_liege = {
				liege = scope:liege
				change = scope:change
			}
			scope:county = {
				change_title_holder = {
					holder = $ACTOR$
					change = scope:change
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		# Uncomment if considering using AGOT's default ruins system
		#if = {
		#	limit = { $PROVINCE$ = { has_holding_type = settlement_holding } }
			scope:county = {
				set_county_faith = $ACTOR$.faith
				if = {
					limit = { culture = culture:unknown_culture }
					if = {
						limit = {
							$ACTOR$.culture = culture:high_valyrian
							has_game_rule = agot_hv_conversion_offshoots
							title_province = { geographical_region = world_westeros }
							NOT = { title_province = { geographical_region = world_valyria_mainland } }
						}
						set_county_culture = culture:westerosi_valyrian
					}
					else_if = {
						limit = {
							$ACTOR$.culture = culture:high_valyrian
							has_game_rule = agot_hv_conversion_offshoots
							NOT = { title_province = { geographical_region = world_westeros } }
							NOT = { title_province = { geographical_region = world_valyria_mainland } }
						}
						set_county_culture = culture:essosi_valyrian
					}
					else = { set_county_culture = $ACTOR$.culture }
				}
			}
			increase_variable = {
				NAME = num_settled_wilderness
				AMOUNT = 1
			}
			trigger_event = agot_colonization_events.0007
			correct_wilderness_tracker = yes
		#}
	}
}

lv_downgrade_settlement_progress_effect = {
	random_list = {
		10 = { # Destroy settlement buildings
			trigger = {
				OR = {
					has_building_or_higher = stockades_01
					has_building_or_higher = local_militia_01
					has_building_or_higher = trading_outposts_01
					has_building_or_higher = construction_guilds_01
					has_building_or_higher = tribal_stockades_01
					has_building_or_higher = tribal_local_militia_01
					has_building_or_higher = tribal_trading_outposts_01
					has_building_or_higher = tribal_construction_guilds_01
					has_building_or_higher = colony_port_01
					has_building_or_higher = colony_quarries_01
					has_building_or_higher = colony_forestry_01
				}
			}
			# Guaranteed one building destroyed
			lv_destroy_random_settlement_building_effect = yes
			random = { # 25% chance to destroy another
				chance = 25
				lv_destroy_random_settlement_building_effect = yes
			}
			random = { # And another
				chance = 5
				lv_destroy_random_settlement_building_effect = yes
			}
		}
		5 = { # Reset blocker progress
			trigger = {
				OR = {
					has_building_or_higher = unexplored_caves_02
					has_building_or_higher = swamp_01_02
					has_building_or_higher = swamp_02_02
					has_building_or_higher = lv_floods_02
					has_building_or_higher = noxious_gases_01_02
					has_building_or_higher = noxious_gases_02_02
					has_building_or_higher = stonemen_encampment_02
					has_building_or_higher = bandit_camps_02
					has_building_or_higher = small_ruined_holding_02
					has_building_or_higher = medium_ruined_holding_02
					has_building_or_higher = large_ruined_holding_02
				}
			}
			lv_reset_random_settlement_blocker_effect = yes
			random = { # Evil
				chance = 5
				lv_reset_random_settlement_blocker_effect = yes
			}
		}
		1 = { # Reset mine progress
			trigger = {
				OR = {
					has_building_or_higher = valyria_mines_dormant_02
					has_building_or_higher = valyria_mines_02
				}
			}
			if = {
				limit = { has_building_or_higher = valyria_mines_dormant_02 }
				remove_building = valyria_mines_dormant_02
				hidden_effect = { add_building = valyria_mines_dormant_01 }
			}
			if = {
				limit = { has_building_or_higher = valyria_mines_02 }
				remove_building = valyria_mines_02
				hidden_effect = { add_building = valyria_mines_01 }
			}
		}
		1 = { # Reset holding building progress
			trigger = { has_building_or_higher = settlement_02 }
			if = {
				limit = { has_building = settlement_03 }
				remove_building = settlement_03
				hidden_effect = { add_building = settlement_02 }
			}
			else = {
				remove_building = settlement_02
				hidden_effect = { add_building = settlement_01 }
			}
		}
	}
}
