## General blocker effects
## All assume province scope
lv_add_blocker_effect = {
	if = {
		limit = { NOT = { has_free_building_slot = yes } }
		if = {
			limit = { always = $FORCE$ }
			add_province_modifier = extra_building_slot
			add_building = $BLOCKER_BUILDING$
			lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = yes }
		}
	}
	else = {
		add_building = $BLOCKER_BUILDING$
		lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = yes }
	}
}

lv_blocker_cleared_effect = {
	lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = no }
	if = { # If AI, try to upgrade settlement once all blockers are cleared
		limit = { NOT = { has_building_with_flag = obstacle } }
		save_scope_as = ruin
		province_owner = {
			if = {
				limit = { is_ai = yes }
				# Trigger 
			}
		}
	}
}

lv_valyria_update_subregion_progress_variables = {
	county = { save_scope_as = county }
	situation:valyria_situation = {
		if = {
			limit = { exists = var:subregion_$SUBREGION$_total }
			if = {
				limit = { always = $ADD_BLOCKER$ }
				change_variable = {
					name = subregion_$SUBREGION$_total
					add = 1
				}
				change_variable = {
					name = subregion_$SUBREGION$_count
					add = 1
				}
			}
			else = {
				change_variable = {
					name = subregion_$SUBREGION$_count
					subtract = 1
				}
			}
		}
		else_if = {
			# Make sure we don't re-add variables to already-rehabilitated regions
			limit = { situation_sub_region:$SUBREGION$ = { NOT = { sub_region_current_phase = valyria_rehabilitated } } }
			set_variable = {
				name = subregion_$SUBREGION$_total
				value = 1
			}
			set_variable = {
				name = subregion_$SUBREGION$_count
				value = 1
			}
		}
		var:subregion_$SUBREGION$_total = { save_scope_as = subregion_total }
		var:subregion_$SUBREGION$_count = { save_scope_as = subregion_count }
		situation_sub_region:$SUBREGION$ = {
			if = { # INSTANT
				limit = { has_game_rule = lv_green_valyria_instant }
				if = {
					limit = { lv_valyria_subregion_progress < 25 }
					change_phase = { phase = valyria_ruined }
				}
				else_if = {
					limit = { lv_valyria_subregion_progress < 50 }
					change_phase = { phase = valyria_restored_01 }
				}
				else_if = {
					limit = { lv_valyria_subregion_progress < 75 }
					change_phase = { phase = valyria_restored_02 }
				}
				else = { change_phase = { phase = valyria_rehabilitated } }
			}
			else_if = { # GRADUAL
				limit = { has_game_rule = lv_green_valyria_gradual }
				if = {
					limit = {
						sub_region_current_phase = valyria_ruined
						lv_valyria_subregion_progress >= 25
						lv_valyria_subregion_progress < 50
					}
					change_phase = { phase = valyria_ruined_transition }
				}
				else_if = {
					limit = {
						sub_region_current_phase = valyria_restored_01
						lv_valyria_subregion_progress >= 50
						lv_valyria_subregion_progress < 75
					}
					change_phase = { phase = valyria_restored_01_transition }
				}
				else_if = {
					limit = {
						sub_region_current_phase = valyria_restored_02
						lv_valyria_subregion_progress >= 75
					}
					change_phase = { phase = valyria_restored_02_transition }
				}
			}
		}
	}
}

lv_blocker_generation_predefined_ruined_holdings_effect = {
	if = { # Large
		limit = {
			OR = {
				county = title:c_valyria
				county = title:c_ilvonys

				county = title:c_oreos
			}
		}
		lv_add_blocker_effect = {
			FORCE = yes
			BLOCKER_BUILDING = large_ruined_holding_01
		}

	else_if = { # Medium
		limit = {
			OR = {
				lv_is_fourteen_flames = yes
				has_special_building = yes
				has_special_building_slot = yes

					any_neighboring_county = {
						any_county_province = {
							OR = {

								county = title:c_ilvonys
								county = title:c_kastaryia

			}
		}

			FORCE = yes
			BLOCKER_BUILDING = medium_ruined_holding_01
		}
	}
	else_if = {
		limit = {
			OR = {
				empire ?= title:e_valyria
				county = title:c_dravathor
				county = title:c_volandrys
				county = {
					any_neighboring_county = {
						OR = {
							any_county_province = {
								lv_is_fourteen_flames = yes
								has_special_building = yes
								has_special_building_slot = yes

		lv_add_blocker_effect = {
			FORCE = yes
			BLOCKER_BUILDING = small_ruined_holding_01

# Random blockers
lv_blocker_generation_random_noxious_gases_effect = {
	if = {
		limit = { NOT = { has_building_or_higher = noxious_gases_02_01 } }

		# Volcanic fumes (noxious_gases_02_)
		random = {
			chance = {
				value = 15
				county = {
					if = { # If province borders a volcano, 90% chance of having level 2 noxious gases
						limit = { any_neighboring_county = { any_county_province = { lv_is_fourteen_flames = yes } } }
						multiply = 6
					}
					else_if = { # If province borders a province that borders a volcano, 30% chance of having level 2 noxious gases
						limit = { any_neighboring_county = { any_neighboring_county = { any_county_province = { lv_is_fourteen_flames = yes } } } }
						multiply = 2
					}
				}
			}
			lv_add_blocker_effect = {
				FORCE = no
				BLOCKER_BUILDING = noxious_gases_02_01
			}
		}
		# Lingering fumes (noxious_gases_01_)
		if = {
			limit = { NOT = { has_building_or_higher = noxious_gases_02_01 } }
			random = {
				chance = {
					value = 15
					county = {
						if = { # Guarantee if neighbor of one of the Fourteen Flames doesn't have the volcanic fumes
							limit = { any_neighboring_county = { any_county_province = { lv_is_fourteen_flames = yes } } }
							value = 100
						}
						else_if = { # If province borders a province that borders a volcano, 90% chance of having level 1 noxious gases
							limit = { any_neighboring_county = { any_neighboring_county = { any_county_province = { lv_is_fourteen_flames = yes } } } }
							multiply = 6
						}
						else = { multiply = 2 } # Everywhere else has a 30% chance of having level 1 noxious gases
					}
				}
				lv_add_blocker_effect = {
					FORCE = no
					BLOCKER_BUILDING = noxious_gases_01_01
				}
			}
		}
	}
}
