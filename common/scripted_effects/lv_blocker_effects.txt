# Legacy of Valyria blocker scripted effects
# Add colony blocker
lv_add_blocker_effect = {
	if = {
		limit = { NOT = { has_free_building_slot = yes } }
		if = {
			limit = { always = $FORCE$ }
			add_province_modifier = extra_building_slot
			add_building = $BLOCKER_BUILDING$
			lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = yes }
		}
	}
	else = {
		add_building = $BLOCKER_BUILDING$
		lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = yes }
	}
}

lv_blocker_cleared_effect = {
	lv_valyria_update_subregion_progress_effect = { ADD_BLOCKER = no }
	if = { # If AI, try to upgrade settlement once all blockers are cleared
		limit = { NOT = { has_building_with_flag = wilderness_blocker } }
		save_scope_as = ruin
		province_owner = {
			if = {
				limit = { is_ai = yes }
				# Trigger 
			}
		}
	}
}

lv_valyria_update_subregion_progress_variables = {
	county = { save_scope_as = county }
	situation:valyria_situation = {
		if = {
			limit = { exists = var:subregion_$SUBREGION$_total }
			if = {
				limit = { always = $ADD_BLOCKER$ }
				change_variable = {
					name = subregion_$SUBREGION$_total
					add = 1
				}
				change_variable = {
					name = subregion_$SUBREGION$_count
					add = 1
				}
			}
			else = {
				change_variable = {
					name = subregion_$SUBREGION$_count
					subtract = 1
				}
			}
		}
		else_if = {
			# Make sure we don't re-add variables to already-rehabilitated regions
			limit = { situation_sub_region:$SUBREGION$ = { NOT = { sub_region_current_phase = valyria_rehabilitated } } }
			set_variable = {
				name = subregion_$SUBREGION$_total
				value = 1
			}
			set_variable = {
				name = subregion_$SUBREGION$_count
				value = 1
			}
		}
		var:subregion_$SUBREGION$_total = { save_scope_as = subregion_total }
		var:subregion_$SUBREGION$_count = { save_scope_as = subregion_count }
		situation_sub_region:$SUBREGION$ = {
			if = {
				limit = { lv_valyria_subregion_progress < 25 }
				change_phase = { phase = valyria_ruined }
			}
			else_if = {
				limit = { lv_valyria_subregion_progress < 50 }
				change_phase = { phase = valyria_restored_01 }
			}
			else_if = {
				limit = { lv_valyria_subregion_progress < 75 }
				change_phase = { phase = valyria_restored_02 }
			}
			else = { change_phase = { phase = valyria_rehabilitated } }
		}
	}
}
