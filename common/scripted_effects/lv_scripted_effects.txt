# Free City-fier effect
lv_free_city_setup_effect = {
	$KINGDOM_TITLE$ = {
		if = {
			limit = { exists = holder }
			holder = {
				save_scope_as = new_ruler
				title:c_myr = { save_scope_as = capital }
				trigger_event = {
					id = free_city_events.9996
					days = 1
				}
				every_vassal_or_below = {
					limit = {
						highest_held_title_tier >= tier_county
						NOT = {
							this = title:d_the_red_priesthood.holder
							this = title:c_temple_of_the_lord_of_light.holder
						}
					}
					if = {
						limit = { has_dlc_feature = admin_gov }
						change_government = administrative_government
					}
					else = { change_government = republic_government }
				}
			}
			set_variable = {
				name = years_since_election
				value = title_held_years
			}
		}
	}
}

lv_dummy_lobotomizer_effect = {
	save_scope_as = dummy
	remove_short_term_gold = current_gold_value
	add_prestige = {
		value = prestige
		multiply = -1
	}
	add_piety = {
		value = piety
		multiply = -1
	}
	every_claim = { scope:dummy = { remove_claim = prev } }
	if = {
		limit = { any_held_title = { tier = tier_barony } }
		every_held_title = {
			limit = { tier = tier_barony }
			title_province = {
				if = {
					limit = {
						is_county_capital = no
						has_holding_type = wilderness_holding
					}
					remove_holding = yes
				}
			}
		}
	}
	if = { # Execute Order 66
		limit = { any_vassal = {} }
		every_realm_province = {
			limit = {
				is_county_capital = no
				has_holding_type = wilderness_holding
			}
			province_owner = { death = natural }
			scope:dummy = { get_title = prev.barony }
			remove_holding = yes
		}
	}
	if = { # Break free
		limit = { is_independent_ruler = no }
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
			add_claim_on_loss = no
		}
		becomes_independent = { change = scope:change }
		resolve_title_and_vassal_change = scope:change
	}
	if = { # Immediate nuke if any title greater than county
		limit = {
			highest_held_title_tier > tier_county
			trigger_if = {
				limit = { has_trait = ruin }
				any_held_title = {
					tier > tier_county
					NOT = { this = title:d_ruins }
				}
			}
			trigger_else = { always = yes }
		}
		if = { # Regain control if already given county titles
			limit = { any_vassal = { highest_held_title_tier = tier_county } }
			every_vassal_or_below = {
				limit = { highest_held_title_tier = tier_county }
				every_held_title = { scope:dummy = { get_title = prev } }
				death = natural
			}
		}
		every_held_title = {
			limit = {
				tier > tier_county
				NOT = { this = title:d_ruins }
			}
			scope:dummy = { destroy_title = prev }
		}
		if = {
			limit = { has_trait = wilderness }
			character:Wilderness_Empress ?= {
				if = {
					limit = {
						is_alive = yes
						this = scope:dummy
					}
					if = {
						limit = { NOT = { has_title = title:c_wilderness } }
						get_title = title:c_wilderness
					}
					set_primary_title_to = title:c_wilderness
				}
				else = {
					scope:dummy = {
						if = {
							limit = { has_title = title:c_wilderness }
							set_primary_title_to = title:c_wilderness
						}
					}
				}
			}
			every_realm_province = {
				limit = {
					is_county_capital = no
					has_holding_type = wilderness_holding
				}
				remove_holding = yes
			}
			change_government = wilderness_government
		}
		else_if = {
			limit = { has_trait = ruin }
			character:Ruin_Empress ?= {
				if = {
					limit = {
						is_alive = yes
						this = scope:dummy
					}
					if = {
						limit = { NOT = { has_title = title:c_ruins } }
						get_title = title:c_ruins
					}
					set_primary_title_to = title:c_ruins
				}
				else = {
					scope:dummy = {
						if = {
							limit = { has_title = title:d_ruins }
							set_primary_title_to = title:d_ruins
						}
						else_if = {
							limit = { has_title = title:c_ruins }
							set_primary_title_to = title:c_ruins
						}
					}
				}
			}
			every_realm_province = {
				limit = {
					is_county_capital = no
					has_holding = yes
				}
				set_holding_type = ruin_holding
			}
			change_government = ruins_government
		}
	}
	if = { # Execute Order 66 (Part 2)
		limit = { any_courtier = {} }
		every_courtier = { death = natural }
	}
	# If, somehow, a rogue ruin character owns a fully-built holding, give away
	if = {
		limit = {
			any_held_title = {
				title_province = {
					is_county_capital = yes
					NOR = {
						has_holding_type = wilderness_holding
						has_holding_type = ruin_holding
					}
				}
			}
		}
		every_held_county = {
			limit = {
				title_province = {
					is_county_capital = yes
					NOR = {
						has_holding_type = wilderness_holding
						has_holding_type = ruin_holding
					}
					save_temporary_scope_as = dummy_distribution_province
				}
			}
			save_scope_as = dummy_distribution_title
			scope:dummy = {
				if = {
					limit = { faith = faith:fg_unknown }
					if = {
						limit = { any_neighboring_and_across_water_top_liege_realm_owner = { NOT = { faith = faith:fg_unknown } } }
						random_neighboring_and_across_water_top_liege_realm_owner = { faith = { save_scope_as = neighbor_faith } }
						set_county_faith = scope:neighbor_faith
					}
				}
				if = {
					limit = { culture = culture:unknown_culture }
					if = {
						limit = { any_neighboring_and_across_water_top_liege_realm_owner = { NOT = { culture = culture:unknown_culture } } }
						random_neighboring_and_across_water_top_liege_realm_owner = { culture = { save_scope_as = neighbor_culture } }
						set_county_culture = scope:neighbor_culture
					}
				}
			}
			if = {
				limit = { scope:dummy_distribution_province.culture = culture:lakefolk }
				create_character = {
					location = scope:dummy_distribution_province
					random_traits = yes
					culture = scope:dummy_distribution_province.culture
					faith = scope:dummy_distribution_province.faith
					age = { 16 55 }
					gender = female
					dynasty = generate
					after_creation = {
						get_title = scope:dummy_distribution_title
						add_gold = { minor_gold_value_check medium_gold_value_check }
						add_prestige = { minor_prestige_gain medium_prestige_gain }
						add_piety = { minor_piety_gain medium_piety_gain }
					}
				}
			}
			else = {
				create_character = {
					location = scope:dummy_distribution_province
					random_traits = yes
					culture = scope:dummy_distribution_province.culture
					faith = scope:dummy_distribution_province.faith
					age = { 16 55 }
					gender = male
					dynasty = generate
					after_creation = {
						get_title = scope:dummy_distribution_title
						add_gold = { minor_gold_value_check medium_gold_value_check }
						add_prestige = { minor_prestige_gain medium_prestige_gain }
						add_piety = { minor_piety_gain medium_piety_gain }
					}
				}
			}
		}
		if = {
			limit = { is_landed = no }
			death = natural
		}
	}
}

# Savi's effect
lv_adventurer_colonize = {
	primary_title = {
		save_scope_as = old_title
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = change
	}

	every_in_list = {
		list = colonized_titles
		limit = {
			tier >= tier_county
			title_province = {
				has_holding_type = ruin_holding
			}
		}
		change_title_holder = {
			holder = scope:new_holder
			change = scope:change
			take_baronies = yes
		}
		title_province = {
			set_holding_type = settlement_holding
		}

		set_county_faith = scope:new_holder.faith
		set_county_culture = scope:new_holder.culture
	}

	resolve_title_and_vassal_change = scope:change
	scope:new_holder = { change_government = feudal_government }
	destroy_laamp_effect = { ADVENTURER = scope:new_holder }

	set_realm_capital = scope:target_county
	if = {
		limit = { exists = scope:target_duchy }
		scope:target_county = { set_coa = scope:old_title }
		scope:target_duchy = { set_coa = scope:old_title }
		get_title = scope:target_duchy
	}
	else = {
		scope:target_county = { set_coa = scope:old_title }
	}
}
