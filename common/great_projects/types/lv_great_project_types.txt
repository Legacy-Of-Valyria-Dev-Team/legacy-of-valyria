			contributor_is_valid = { highest_held_title_tier > tier_barony }

			# Can only pay for this or the extensive option
			is_shown = {
				trigger_if = {
					limit = { exists = scope:great_project }
					scope:great_project ?= {
						any_contribution = {
							contribution_id = extensive_artifact_surveys
							contribution_is_funded = no
						}
					}
				}
				trigger_else = { always = yes }
			}

			cost = { treasury_or_gold = { value = great_project_base_contribution_cost } }

			contributor_cooldown = individual_contribution_cooldown_value

			is_required = no

			on_complete = {
				great_project_reward_contribution_completion_notification_effect = {
					LEVEL = minor
					NON_OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
					OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
				}
				custom_tooltip = {
					text = great_project_type_lv_rehabilitate_valyria_contribution_minimal_artifact_surveys_tooltip
					trigger_event = lv_great_project.0006
				}
			}
			on_contribution_funded = { great_project_notify_contribution = yes }

			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 9
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = chinese_great_projects_contribution_ai_will_do_base_value }
		}
		extensive_artifact_surveys = {
			contributor_is_valid = { highest_held_title_tier > tier_barony }

			cost = {
				treasury_or_gold = {
					value = great_project_base_contribution_cost
					multiply = 3
				}
			}

			contributor_cooldown = individual_contribution_cooldown_value

			is_required = no

			on_complete = {
				great_project_reward_contribution_completion_notification_effect = {
					LEVEL = major
					NON_OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
					OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
				}
				custom_tooltip = {
					text = great_project_type_lv_rehabilitate_valyria_contribution_extensive_artifact_surveys_tooltip
					trigger_event = lv_great_project.0006
				}
			}
			on_contribution_funded = {
				great_project_notify_contribution = yes

				# Pseudo-refund since we can't refund contributions
				scope:great_project ?= {
					if = {
						limit = {
							any_contribution = {
								contribution_id = minimal_artifact_surveys
								contribution_is_funded = yes
								save_temporary_scope_as = artifact_contribution
							}
						}
						scope:artifact_contribution ?= {
							contributor = {
								send_interface_toast = {
									type = event_toast_effect_neutral
									title = minimal_artifact_surveys_costs_refunded_title
									left_icon = root
									right_icon = scope:owner
									custom_tooltip = minimal_artifact_surveys_costs_refunded_desc
									add_gold = great_project_base_contribution_cost
								}
							}
						}
					}
				}
			}

			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 9
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = chinese_great_projects_contribution_ai_will_do_base_value }
		}
	}

	ai_target_quick_trigger = {
		adult = yes
		rank = empire
	}
	ai_check_interval_by_tier = {					
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 6
		hegemony = 6
	}
	ai_will_do = {
		value = 30
		if = {
			limit = { treasury >= 20000 }
			add = 70
		}
		if = {
			limit = { is_at_war = yes }
			multiply = 0
		}
	}

	completion_sound_effect = @msg_completion_effect_generic
}
