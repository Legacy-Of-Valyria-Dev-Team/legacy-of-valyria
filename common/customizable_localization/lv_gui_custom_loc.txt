# Legacy of Valyria custom GUI locs

# Begin Expedition
LV_BeginExpedition_Desc = {
	type = landed_title

	# Special Locations
	text = {
		trigger = { this = title:c_yeen }
		localization_key = lv_begin_expedition_desc_sothoryos_yeen
	}
	text = {
		trigger = { this = title:c_zamettar }
		localization_key = lv_begin_expedition_desc_sothoryos_zamettar
	}
	text = {
		trigger = { this = title:c_gorosh }
		localization_key = lv_begin_expedition_desc_sothoryos_gorosh
	}
	# Region-Specific Text
	text = { # Lands of the Long Summer - Wilderness
		trigger = {
			title_province = {
				has_holding_type = wilderness_holding
				geographical_region = world_valyria_lands_of_the_long_summer
			}
		}
		localization_key = lv_begin_expedition_desc_lands_of_the_long_summer_wilderness
	}
	text = { # Valyria - Ruins
		trigger = {
			title_province = {
				has_holding_type = ruin_holding
				geographical_region = world_valyria_proper
			}
		}
		localization_key = lv_begin_expedition_desc_valyria_ruin
	}
	text = { # Valyria - Wilderness
		trigger = {
			title_province = {
				has_holding_type = wilderness_holding
				geographical_region = world_valyria_proper
			}
		}
		localization_key = lv_begin_expedition_desc_valyria_wilderness
	}
	text = { # Sothoryos - Wilderness
		trigger = {
			title_province = {
				has_holding_type = wilderness_holding
				geographical_region = world_sothoryos
			}
		}
		localization_key = lv_begin_expedition_desc_sothoryos_wilderness
	}
	text = {
		trigger = { title_province = { has_holding_type = ruin_holding } }
		localization_key = lv_begin_expedition_desc_ruin
	}
	text = {
		trigger = { title_province = { has_holding_type = wilderness_holding } }
		localization_key = lv_begin_expedition_desc_wilderness
	}
	text = {
		fallback = yes
		localization_key = lv_begin_expedition_desc_fallback
	}
}

# Wild Dragons View Widget
LV_WildDragonsView_Name = {
	type = character

	text = { # Enraged Firewyrm
		trigger = {
			has_trait = firewyrm
			any_owned_story = {
				story_type = lv_firewyrm_story_cycle
				var:lv_firewyrm_mood ?= flag:enraged
			}
		}
		localization_key = lv_wild_dragons_view_name_firewyrm_enraged
	}
	text = { # Agitated Firewyrm
		trigger = {
			has_trait = firewyrm
			any_owned_story = {
				story_type = lv_firewyrm_story_cycle
				var:lv_firewyrm_mood ?= flag:agitated
			}
		}
		localization_key = lv_wild_dragons_view_name_firewyrm_agitated
	}
	text = { # Firewyrm
		trigger = { has_trait = firewyrm }
		localization_key = lv_wild_dragons_view_name_firewyrm
	}
	text = {
		fallback = yes
		localization_key = lv_wild_dragons_view_name_dragon
	}
}
