	##################################################

	on_monthly = {
		every_situation_sub_region = { # Do monthly events
			# Regular monthly pulse
			every_situation_sub_region_county = {
				limit = { holder = { agot_is_dummy_character = no } }
				save_scope_as = county

				# Set stewardship buff
				if = {
					limit = {
						title_province = { has_holding_type = settlement_holding }
						is_target_of_council_task = task_develop_county
					}
					holder = {
						random_stewardship_councillor = {
							limit = { is_performing_council_task = task_develop_county }
							save_scope_as = steward
						}
					}
					if = {
						limit = { scope:steward.stewardship >= 11 }
						if = {
							limit = { NOT = { has_county_modifier = lv_restoration_steward_modifier } }
							holder = {
								send_interface_toast = {
									title = lv_restoration_steward_title
									custom_tooltip = lv_restoration_steward_desc
									left_icon = scope:steward
									prev = { add_county_modifier = lv_restoration_steward_modifier }
								}
							}
						}
						else = {
							remove_county_modifier = lv_restoration_steward_modifier
							add_county_modifier = lv_restoration_steward_modifier
						}
					}
				}
				else_if = {
					limit = { has_county_modifier = lv_restoration_steward_modifier }
					remove_county_modifier = lv_restoration_steward_modifier
				}
			}
		}
	}
