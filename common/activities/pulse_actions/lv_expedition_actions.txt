# You discover a half-buried monument
lv_apa_expedition_monument = {
	icon = learning
	is_valid = {
		is_current_phase_active = yes
		# Replace with event chain variable once coded in
	}

	weight = { value = 1 }

	effect = {
		scope:host = { save_scope_as = first }
		scope:province = { culture = { save_scope_as = second } }

		add_activity_log_entry = {
			key = lv_apa_expedition_monument
			tags = { pulse_action }
			character = scope:first
	
			custom_tooltip = lv_expedition.tt.positive.tiny
			scope:activity = { activity_special_type_progression_tiny = yes }
		}
	}
}

## PROGRESS DECREASE
# A batch of your supplies becomes spoiled
lv_apa_expedition_spoiled_supplies = {
	icon = learning
	is_valid = {
		is_current_phase_active = yes
		# Replace with event chain variable once coded in
	}

	weight = {
		value = 0.5
		if = {
			limit = {
				scope:activity = {
					has_activity_type = lv_activity_expedition
					has_activity_option = {
						category = expedition_preparation
						option = expedition_preparation_standard
					}
				}
			}
			add = -0.4
		}
		else_if = {
			limit = {
				scope:activity = {
					has_activity_type = lv_activity_expedition
					has_activity_option = {
						category = expedition_preparation
						option = expedition_preparation_heavy
					}
				}
			}
			add = -0.45
		}
	}

	effect = {
		scope:host = { save_scope_as = first }

		add_activity_log_entry = {
			key = lv_apa_expedition_spoiled_supplies
			tags = { pulse_action }
			character = scope:first
	
			custom_tooltip = lv_expedition.tt.negative
			scope:activity = { activity_special_type_progression_negative = yes }
		}
	}
}

# A storm turns the nearby roads to muck, stalling the entire expedition
lv_apa_expedition_mired_in_mud = {
	icon = learning
	is_valid = {
		is_current_phase_active = yes
		# Replace with event chain variable once coded in
	}

	weight = {
		value = 0.5
		if = {
			limit = {
				scope:province = {
					OR = {
						agot_is_cloudforest_terrain = yes
						agot_is_jungle_terrain = yes
					}
				}
			}
			add = 0.5
		}
		if = {
			limit = { scope:province = { agot_is_wetlands_terrain = yes } }
			add = 0.4
		}
		if = {
			limit = {
				scope:province = {
					OR = {
						agot_is_the_bog_terrain = yes
						agot_is_forest_terrain = yes
						agot_is_floodplains_terrain = yes
					}
				}
			}
			add = 0.3
		}
		if = {
			limit = {
				scope:activity = {
					has_activity_type = lv_activity_expedition
					has_activity_option = {
						category = expedition_preparation
						option = expedition_preparation_standard
					}
				}
			}
			multiply = 0.67
		}
		else_if = {
			limit = {
				scope:activity = {
					has_activity_type = lv_activity_expedition
					has_activity_option = {
						category = expedition_preparation
						option = expedition_preparation_heavy
					}
				}
			}
			multiply = 0.33
		}
	}

	effect = {
		scope:host = { save_scope_as = first }

		add_activity_log_entry = {
			key = lv_apa_expedition_ruined_roads
			tags = { pulse_action }
			character = scope:first
	
			custom_tooltip = lv_expedition.tt.negative
			scope:activity = { activity_special_type_progression_negative = yes }
		}
	}
}

				}
			}
		}
	}
