# Legacy of Valyria
lv_view_wild_dragons = {
	is_shown = {
		any_character_in_location = {
			has_trait = dragon
			NOT = { has_character_flag = owned_dragon }
		}
	}

	effect = { trigger_event = lv_wild_dragons_view.0001 }
}

lv_view_wild_dragons_agitated_or_enraged_firewyrms = {
	is_shown = {
		any_character_in_location = {
			lv_is_firewyrm = yes
			any_owned_story = {
				story_type = lv_firewyrm_story_cycle
				OR = {
					var:lv_firewyrm_mood ?= flag:agitated
					var:lv_firewyrm_mood ?= flag:enraged
				}
			}
		}
	}
}
