can_reconstruct_ruin = {
	scope = province
	is_valid = {
		save_temporary_scope_as = ruin
		OR = {
			scope:actor = {
				OR = {
					any_character_to_title_neighboring_and_across_water_county = {
						this = scope:ruin.county
					}
					# Ghozai
					trigger_if = {
						limit = {
							title:d_velos = { is_de_jure_liege_or_above_target = scope:ruin.county }
						}
						any_sub_realm_county = {
							is_coastal_county = yes
							title_province = {
								OR = {
									geographical_region = world_essos_painted_mountains
									geographical_region = world_essos_elyria
									geographical_region = world_essos_slavers_bay
								}
							}
						}
					}
					# Manticore Isles
					trigger_if = {
						limit = {
							title:d_manticore_isles = { is_de_jure_liege_or_above_target = scope:ruin.county }
						}
						any_sub_realm_county = {
							is_coastal_county = yes
							title_province = {
								OR = {
									geographical_region = world_asshai
									geographical_region = world_marahai # Formerly world_far_essos_volcanos
								}
							}
						}
					}
				}
			}
			scope:actor = {
				any_held_title = {
					is_de_jure_liege_or_above_target = scope:ruin.county
				}
			}
			scope:actor = {
				trigger_if = {
					limit = {
						is_ai = no
						prestige_level >= 4
						any_directly_owned_province = {
							has_building_or_higher = common_tradeport_06
						}
					}
					scope:ruin.county = {
						is_coastal_county = yes
					}
				}
				trigger_else = { always = no }
			}
		}
		trigger_if = {
			limit = { has_building_or_higher = large_ruin_01 }
			trigger_if = {
				limit = { has_building = large_ruin_03 }
				scope:actor = { gold >= 1800 }
			}
			trigger_else_if = {
				limit = { has_building = large_ruin_02 }
				scope:actor = { gold >= 1200 }
			}
			trigger_else = {
				scope:actor = { gold >= 600 }
			}
		}
		trigger_else_if = {
			limit = { has_building_or_higher = medium_ruin_01 }
			trigger_if = {
				limit = { has_building = medium_ruin_03 }
				scope:actor = { gold >= 1200 }
			}
			trigger_else_if = {
				limit = { has_building = medium_ruin_02 }
				scope:actor = { gold >= 800 }
			}
			trigger_else = {
				scope:actor = { gold >= 400 }
			}
		}
		trigger_else_if = {
			limit = { has_building_or_higher = small_ruin_01 }
			trigger_if = {
				limit = { has_building = small_ruin_03 }
				scope:actor = { gold >= 600 }
			}
			trigger_else_if = {
				limit = { has_building = small_ruin_02 }
				scope:actor = { gold >= 400 }
			}
			trigger_else = {
				scope:actor = { gold >= 200 }
			}
		}
		trigger_else = {
			always = no
		}
	}
}

# LEGACY OF VALYRIA MODIFIED
reconstruct_ruin = {
	scope = province
	effect = {
		scope:actor = {
			if = {
				limit = { prev = { has_building_or_higher = large_ruin_01 } }
				prev = { set_variable = { name = ruin_size value = flag:ruin_size_large } }
				remove_short_term_gold = 600
			}
			else_if = {
				limit = { prev = { has_building_or_higher = medium_ruin_01 } }
				prev = { set_variable = { name = ruin_size value = flag:ruin_size_medium } }
				remove_short_term_gold = 400
			}
			else = {
				remove_short_term_gold = 200
			}
			get_title = prev.county
		}
		# Determine ruin level for converting to appropriate colony building
		if = {
			limit = {
				OR = {
					has_building = large_ruin_03
					has_building = medium_ruin_03
					has_building = small_ruin_03
				}
			}
			set_variable = { name = ruin_progress value = flag:ruin_progress_significant }
		}
		else_if = {
			limit = {
				OR = {
					has_building = large_ruin_02
					has_building = medium_ruin_02
					has_building = small_ruin_02
				}
			}
			set_variable = { name = ruin_progress value = flag:ruin_progress_minimal }
		}
		set_holding_type = colony_holding
		# Convert to appropriate colony level if prior progress made
		if = { # Colony level 3 building
			limit = { var:ruin_progress = flag:ruin_progress_significant }
			add_building = colony_03
		}
		else_if = { # Colony level 2 building
			limit = { var:ruin_progress = flag:ruin_progress_minimal }
			add_building = colony_02
		}
		remove_variable = ruin_progress
	}
}
