# Legacy of Valyria
lv_use_custom_colonization = {
	scope = title
	is_shown = { NOT = { has_game_rule = lv_colonisation_agot } }
}

# Used for restoration cost calculation
lv_restore_ruins_tooltip_show = {
	scope = province
	effect = {
		save_scope_as = ruin
		scope:actor = {
			set_variable = {
				name = lv_restore_ruins_tooltip
				value = scope:ruin
			}
		}
	}
}
lv_restore_ruins_tooltip_hide = {
	scope = province
	effect = { scope:actor = { remove_variable = lv_restore_ruins_tooltip } }
}
lv_restore_ruins_tooltip_culture_valyrian_architecture = {
	is_shown = {
		scope:ruin ?= { county.culture = culture:high_valyrian }
		culture = { has_building_gfx = high_valyrian_building_gfx }
	}
}
lv_restore_ruins_tooltip_culture_hard_working = { is_shown = { culture = { has_cultural_tradition = tradition_hard_working } } }

}

lv_ruin_can_afford = { is_shown = { gold >= lv_settlement_cost } }

lv_ruin_is_unique = {
	scope = province
	is_valid = {
		has_holding_type = ruin_holding
		OR = {
			empire ?= title:e_valyria
			empire ?= title:e_lands_of_long_summer
		}
	}
}

lv_ruin_has_blockers = {
	scope = province
	is_valid = { has_building_with_flag = obstacle }
}
