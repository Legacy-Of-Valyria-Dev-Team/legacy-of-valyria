colonize_county_or_duchy_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_opening.dds" }
	desc = colonize_county_or_duchy_decision_desc
	selection_tooltip = colonize_county_or_duchy_decision_tooltip
	sort_order = 200
	decision_group_type = adventurer_minor
	
	is_shown = {
		has_ep3_dlc_trigger = yes
		is_landless_adventurer = yes
		location = {
			has_holding_type = ruin_holding	# Might have to check for ruin warden as the holder instead to avoid players colonizing someone's province effectively providing a game over again lol
			province_owner = { has_trait = ruin }
		}
	}

	is_valid_showing_failures_only = {
		is_alive = yes
		is_available_adult = yes
		has_contagious_deadly_disease_trigger = no
		root.location = {
			is_occupied = no
			is_raided = no
		}
	}

	is_valid = {
		domicile ?= { has_domicile_building_or_higher = settler_camp_03 }
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "colonize_county_or_duchy_decision_confirm"

		item = { # Settle 1 county
			value = colonize_one_county
			current_description = colonize_one_county_desc
			localization = colonize_one_county
			icon = "gfx/interface/icons/county_crown.dds"
			is_valid = {
				custom_tooltip = {
					text = need_settler_camp_3_or_higher
					domicile = {
						has_domicile_building_or_higher = settler_camp_03
					}
				}
			}
			is_shown = {
				domicile ?= { has_domicile_building_or_higher = settler_camp_03 }
			}
		}
		item = { # Settle the whole duchy
			value = colonize_a_duchy
			current_description = colonize_a_duchy_desc
			localization = colonize_a_duchy
			icon = "gfx/interface/icons/duchy_crown.dds"
			is_valid = {
				custom_tooltip = {
					text = need_settler_camp_6
					domicile = {
						has_domicile_building_or_higher = settler_camp_06
					}
				}
			}
			is_shown = {
				domicile ?= { has_domicile_building_or_higher = settler_camp_06 }
			}
		}
	}

	effect = {
		close_view = {
			view = decisions
			player = root
		}
		custom_tooltip = gain_trait
		custom_tooltip = courtier_gain_trait
		custom_tooltip = settle_location

		hidden_effect = {
			if = {
				limit = { NOT = { has_trait = adventurer } }
				add_trait = adventurer
			}
			every_courtier = {
				set_variable = {
					name = former_camp_leader
					value = root
				}
				add_trait = adventurer_follower
			}

			if = {
				limit = { scope:colonize_one_county = yes }
				root.location.county = { save_temporary_scope_as = target_county }
				root = { save_temporary_scope_as = new_holder }
				scope:target_county = {
					add_to_list = colonized_titles
				}
				lv_adventurer_colonize = yes
				
			}
			else_if = {
				limit = { scope:colonize_a_duchy = yes }
				root.location = {
					county = {
						save_temporary_scope_as = target_county
					}
					duchy = {
						save_temporary_scope_as = target_duchy
					}
				}
				# root.location.county = { save_temporary_scope_as = target_county }
				# root.location.duchy = { save_temporary_scope_as = target_duchy }
				root = { save_temporary_scope_as = new_holder }
	
				scope:target_duchy = {
					add_to_list = colonized_titles
					every_de_jure_county = {
						save_temporary_scope_as = target_counties
						add_to_list = colonized_titles
					}
				}
				lv_adventurer_colonize = yes
			}
		}
	}

	ai_check_interval = 60

	ai_potential = {
		government_has_flag = government_is_landless_adventurer
		is_available = yes
	}

	ai_will_do = {
		base = 50
	}
}
