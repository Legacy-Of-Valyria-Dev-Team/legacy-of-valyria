# Legacy of Valyria major decisions

# Restore the Valyrian Freehold
lv_restore_valyrian_freehold_decision = {
    picture = { reference = "gfx/interface/illustrations/event_scenes/lv_valyria_prospering.dds" }
    desc = lv_restore_valyrian_freehold_decision_desc
    selection_tooltip = lv_restore_valyrian_freehold_decision_tooltip
    decision_group_type = valyria

    ai_check_interval_by_tier = {
        barony = 0
        county = 0
        duchy = 0
        kingdom = 0
        empire = 120
        hegemony = 0
    }

    is_shown = {
        is_ruler = yes
        is_playable_character = yes
        OR = {
            has_title = title:e_valyria
            primary_title.title_capital_county ?= { title_province = { geographical_region = world_valyria_mainland } }
        }
        NOT = {  # Can only do it once.
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_valyrian_freehold
            }
        }
        NOT = { exists = title:h_valyrian_freehold.holder }
        NOT = { government_has_flag = government_is_nw }
    }

    is_valid = {
        prestige_level >= max_prestige_level

        custom_tooltip = {
            text = lv_restore_valyrian_freehold_valyria_requirement_tooltip
            completely_controls = title:e_valyria
            completely_controls = title:e_lands_of_long_summer
        }
        custom_tooltip = {
            text = lv_restore_valyrian_freehold_former_colonies_requirement_tooltip
            any_empire = {
                OR = {
                    this = title:e_old_valyria
                    this = title:e_ghiscar
                    this = title:e_volantis
                    this = title:e_rhoyne
                    this = title:e_three_daughters
                    this = title:e_pentos
                }
                save_temporary_scope_as = former_colony
                root = {
                    OR = {
                        completely_controls = scope:former_colony
                        trigger_if = {
                            limit = { scope:former_colony = title:e_ghiscar }
                            completely_controls = title:k_meereen
                            completely_controls = title:k_yunkai
                            completely_controls = title:k_astapor
                            completely_controls = title:d_new_ghis # To bypass need for Old Ghis
                        }
                        trigger_else_if = {
                            limit = { scope:former_colony = title:e_volantis }
                            completely_controls = title:d_volantis # To bypass need for Rhoynar ruins (probably add more checks since Volantis is massive)
                        }
                        trigger_else_if = {
                            limit = { scope:former_colony = title:e_three_daughters }
                            completely_controls = title:k_tyrosh
                            completely_controls = title:k_lys
                            completely_controls = title:k_myr
                        }
                        trigger_else = { always = no }
                    }
                }
                is_titular = no
                count >= 3
            }
        }
    }

    is_valid_showing_failures_only = {
        is_landed = yes
        is_capable_adult = yes
        is_imprisoned = no
        top_liege = this

        OR = {
            religion = religion:the_flames_religion
            culture = { has_cultural_pillar = heritage_valyrian }
        }
    }

    confirm_text = lv_restore_valyrian_freehold_decision_confirm

    effect = {
        custom_tooltip = lv_restore_valyrian_freehold_decision_effect_tooltip
        show_as_tooltip = { lv_restore_valyrian_freehold_effect = yes }
        trigger_event = lv_valyrian_freehold.0001
        hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_valyrian_freehold
            }
        }
        root.culture = {
            if = {
                limit = { has_cultural_tradition = lv_tradition_north_valyrian_1 }
                remove_culture_tradition = lv_tradition_north_valyrian_1
            }
			add_culture_tradition = lv_tradition_north_valyrian_2
        }
    }

    ai_potential = { always = yes }

    ai_will_do = { base = 100 }
}
